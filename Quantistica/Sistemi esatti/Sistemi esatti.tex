\section{Particella Libera 1D} %

Consideriamo una particella libera di muoversi in una dimensione, senza forze applicate: %
\begin{equation*}
    \boxed{ \hat{V} = 0 } \quad \Rightarrow \quad \hat{H} = -\frac{\hbar^2}{2m} \frac{\partial^2}{\partial x^2} %
\end{equation*}

\noindent
Cerchiamo gli autostati dell'energia risolvendo l'equazione indipendente dal tempo $\hat{H}u = Eu$: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{\partial^2 u}{\partial x^2} = E u \quad \Rightarrow \quad \frac{\partial^2 u}{\partial x^2} = -\frac{2mE}{\hbar^2} u %
\end{equation*}
Definendo $k^2 = \frac{2mE}{\hbar^2}$, otteniamo: %
\begin{equation}
    \frac{\partial^2 u}{\partial x^2} = -k^2 u \quad \rightarrow \quad \text{Eq. oscillatore armonico } \left(\frac{d^2 x}{dt^2} = -\omega^2 x\right) %
\end{equation}
La soluzione generale (con $E$ fissato) è: %
\begin{equation*}
    u(x) = A \cos(kx) + B \sin(kx) \quad \text{con } k \in [0, +\infty] %
\end{equation*}
$\hookrightarrow$ Abbiamo trovato una \textbf{famiglia di soluzioni}. Poiché $k$ è continuo, lo spettro dell'energia è \textbf{continuo}: $E = \frac{\hbar^2 k^2}{2m}$ %.

\noindent
Utilizzando le formule di Eulero, possiamo riscrivere la soluzione con gli esponenziali complessi: %
\begin{align*}
    u(x) &= A \frac{e^{ikx} + e^{-ikx}}{2} + B \frac{e^{ikx} - e^{-ikx}}{2i} = \left( \frac{A}{2} + \frac{B}{2i} \right) e^{ikx} + \left( \frac{A}{2} - \frac{B}{2i} \right) e^{-ikx} = \\ %
         &= C e^{ikx} + D e^{-ikx} \quad \text{(onda progressiva + onda regressiva)} %
\end{align*}

\noindent
Reintroducendo la dipendenza temporale per ottenere gli \textbf{Stati Stazionari} ($\Psi = u(x) e^{-i\frac{E}{\hbar}t}$ con $\omega = \frac{E}{\hbar}$): %
\begin{equation}
    \Psi(x,t) = C e^{i(kx - \omega t)} + D e^{i(-kx - \omega t)} %
\end{equation}
Questo stato ha un'energia esattamente determinata $E = \frac{\hbar^2 k^2}{2m}$ (è un Autostato dell'operatore $\hat{H}$) %.

\subsection{Cosa possiamo dire del Momento $\hat{p}$?} %

Risolviamo l'equazione agli autovalori per l'operatore momento $\hat{p} u_p = p u_p$: %
\begin{equation*}
    -i\hbar \frac{\partial u_p}{\partial x} = p u_p \quad \Rightarrow \quad u_p(x) = c e^{i\frac{p}{\hbar}x} %
\end{equation*}
Con $p \in [-\infty, +\infty] \rightarrow$ \textbf{Spettro Continuo}. %

\noindent
\textbf{Normalizzazione alla Dirac:} Per gli spettri continui, imponiamo $\langle u_p | u_{p'} \rangle = \delta(p - p')$ %:
\begin{align*}
    \langle u_p | u_{p'} \rangle &= \int_{\mathbb{R}} u_p^*(x) u_{p'}(x) \, dx = \int_{\mathbb{R}} c^* e^{-i\frac{p}{\hbar}x} c e^{i\frac{p'}{\hbar}x} \, dx = \\ %
    &= |c|^2 \underbrace{ \int_{\mathbb{R}} e^{i\frac{(p' - p)}{\hbar}x} \, dx }_{2\pi\hbar \, \delta(p' - p)} = |c|^2 2\pi\hbar \, \delta(p' - p) %
\end{align*}
Uguagliando a $\delta(p - p')$, otteniamo $|c|^2 = \frac{1}{2\pi\hbar} \Rightarrow c = \frac{1}{\sqrt{2\pi\hbar}}$ %. L'autostato normalizzato del momento è: %
\begin{equation}
    u_p(x) = \frac{1}{\sqrt{2\pi\hbar}} e^{i\frac{p}{\hbar}x} %
\end{equation}

Confrontiamo gli autostati: %
\begin{align*}
    \hat{H} \quad &\Rightarrow \quad u(x) = C e^{ikx} + D e^{-ikx} = C e^{i\frac{p}{\hbar}x} + D e^{-i\frac{p}{\hbar}x} \quad \text{(Autostati)} \\ %
    \hat{p} \quad &\Rightarrow \quad u_p(x) = \frac{1}{\sqrt{2\pi\hbar}} e^{i\frac{p}{\hbar}x} %
\end{align*}
\begin{itemize}
    \item Se $C \neq 0$ e $D = 0 \quad \Rightarrow \quad u(x) \propto u_p(x)$ (Onda progressiva) %
    \item Se $C = 0$ e $D \neq 0 \quad \Rightarrow \quad u(x) \propto u_{-p}(x)$ (Onda regressiva) %
\end{itemize}

Questo accade perché \textbf{Hamiltoniana e Momento commutano} per la particella libera: %
\begin{equation*}
    [\hat{p}_x, \hat{H}] = [\hat{p}_x, \hat{K}(\hat{p}_x)] = 0 \quad \Rightarrow \quad \text{Hanno almeno 1 base in comune} %
\end{equation*}

\subsection{Esempio Pratico (Sovrapposizione di Momenti)} %

Sia dato lo stato stazionario (autostato di $\hat{H}$): %
\begin{equation*}
    u(x) = \frac{1}{\sqrt{3}} e^{ik_0 x} + \sqrt{\frac{2}{3}} e^{-ik_0 x} %
\end{equation*}
Esso possiede un'energia ben determinata $E_0 = \frac{\hbar^2 k_0^2}{2m}$ %.

\noindent
Applichiamo l'operatore momento $\hat{p}$: %
\begin{equation*}
    \hat{p} u(x) = -i\hbar \frac{\partial}{\partial x} \left[ \frac{1}{\sqrt{3}} e^{ik_0 x} + \sqrt{\frac{2}{3}} e^{-ik_0 x} \right] = \underbrace{\hbar k_0}_{p} \left[ \frac{1}{\sqrt{3}} e^{ik_0 x} - \sqrt{\frac{2}{3}} e^{-ik_0 x} \right] \neq p \cdot u(x) %
\end{equation*}
$\Rightarrow u(x)$ \textbf{NON È un autostato di $\hat{p}$}. %

\noindent
Tuttavia, possiamo esprimerlo come combinazione lineare degli autostati di $\hat{p}$: %
\begin{equation*}
    u(x) = \alpha u_{p_0}(x) + \beta u_{-p_0}(x) \quad \text{dove } u_{\pm p_0}(x) = \frac{1}{\sqrt{2\pi\hbar}} e^{\pm i\frac{p_0}{\hbar}x} %
\end{equation*}
Le probabilità di misurare $p_0$ o $-p_0$ sono i moduli quadri dei coefficienti: %
\begin{align*}
    \mathbb{P}(p = p_0) &= \left| \frac{1}{\sqrt{3}} \right|^2 = \frac{1}{3} \\ %
    \mathbb{P}(p = -p_0) &= \left| \sqrt{\frac{2}{3}} \right|^2 = \frac{2}{3} %
\end{align*}

\noindent
Calcoliamo i valori medi: %
\begin{align*}
    \langle \hat{p} \rangle &= \frac{1}{3}(p_0) + \frac{2}{3}(-p_0) = -\frac{1}{3} p_0 \\ %
    \langle \hat{p}^2 \rangle &= \frac{1}{3}(p_0^2) + \frac{2}{3}(p_0^2) = p_0^2 %
\end{align*}
Da cui l'incertezza sul momento (Deviazione Standard): %
\begin{equation}
    \sigma_p = \Delta p = \sqrt{ \langle \hat{p}^2 \rangle - \langle \hat{p} \rangle^2 } = \sqrt{ p_0^2 - \left(-\frac{1}{3}p_0\right)^2 } = \sqrt{ \frac{8}{9} p_0^2 } %
\end{equation}

\noindent
\textbf{Collasso post-misurazione:} %
Se misuro e \textbf{TROVO} $\mathbf{p_0} \Rightarrow$ la funzione d'onda \textbf{COLLASSA} in $u_p(x) = \frac{1}{\sqrt{2\pi\hbar}} e^{i\frac{p_0}{\hbar}x}$ %.
Adesso il sistema si trova in un autostato sia di $\hat{p}$ che di $\hat{H}$: %
\begin{itemize}
    \item $\sigma_p = 0 \Rightarrow \langle \hat{p} \rangle = p_0$ %
    \item $\sigma_H = 0 \Rightarrow \langle \hat{H} \rangle = \frac{p_0^2}{2m} = E_0$ (L'energia è rimasta ben definita!) %
\end{itemize}

\subsection{Onda Stazionaria Pura} %
Se i coefficienti sono uguali, si crea un'onda stazionaria perfetta: %
\begin{equation*}
    u(x) = \frac{1}{\sqrt{2}} e^{i\frac{p_0}{\hbar}x} + \frac{1}{\sqrt{2}} e^{-i\frac{p_0}{\hbar}x} \quad \Rightarrow \quad \langle \hat{p} \rangle = 0 %
\end{equation*}
$\left(\mathbb{P}(p=p_0) = \frac{1}{2} \text{ e } \mathbb{P}(p=-p_0) = \frac{1}{2} \right)$\\

\noindent
\textbf{Nota finale:} La soluzione fisicamente realistica all'equazione $-\frac{\hbar^2}{2m} \frac{\partial^2 u}{\partial x^2} = Eu$ non è la singola onda piana, ma un pacchetto d'onda ottenuto sovrapponendo infiniti $k$: %
\begin{equation}
    u(x) = \int_{\mathbb{R}} A(k) e^{ikx} \, dk \quad \text{(\textbf{SOLUZ. VERA}, pacchetto d'onda)} %
\end{equation}

\section{Buca di Potenziale Infinita (Particella nella Scatola)} %

    \centering

        \centering
        \includegraphics[width=\linewidth]{immagini/2026-04-09-16-43-51.png}

    \begin{minipage}{0.65\textwidth}
        Consideriamo una particella di massa $m$ vincolata a muoversi in una regione unidimensionale di lunghezza $a$. L'energia potenziale è definita come: %
        \begin{equation}
            V(x) = \begin{cases}
                0 & \text{se } 0 \le x \le a \\ %
                \infty & \text{altrove (fuori dalla buca)} %
            \end{cases}
        \end{equation}
    \end{minipage}
\end{figure}

Risolviamo l'equazione di Schrödinger indipendente dal tempo: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{\partial^2 u}{\partial x^2} + V u = E u %
\end{equation*}

\noindent
\textbf{1. Fuori dalla buca ($x < 0$ e $x > a$):} %
Qui $V \to \infty$. Affinché l'energia totale $E$ rimanga una quantità fisica finita (e non diverga all'infinito), la funzione d'onda deve necessariamente annullarsi. %
\begin{equation*}
    u(x) = 0 \quad \text{per } x < 0 \text{ e } x > a %
\end{equation*}

\noindent
\textbf{2. Condizioni di Continuità ai bordi:} %
Poiché la funzione d'onda $\Psi$ deve essere ovunque continua (Proprietà 3), i limiti ai bordi della buca devono raccordarsi con lo zero esterno: %
\begin{equation}
    u(x=0) = 0 \quad \text{e} \quad u(x=a) = 0 \quad \text{(\textbf{Condizioni al contorno})} %
\end{equation}

\subsection{Soluzione all'interno della Buca ($0 \le x \le a$)} %

Dentro la buca $V=0$, quindi il sistema è identico a quello della particella libera: %
\begin{equation*}
    E = \frac{\hbar^2 k^2}{2m} \quad \Rightarrow \quad k^2 = \frac{2mE}{\hbar^2} \quad \Rightarrow \quad u(x) = A\cos(kx) + B\sin(kx) %
\end{equation*}

Applichiamo ora le condizioni al contorno per trovare le costanti $A$ e $B$: %
\begin{itemize}
    \item \textbf{Condizione in $\mathbf{x=0}$:} %
    \begin{equation*}
        u(0) = A\cos(0) + B\sin(0) = A \cdot 1 + 0 = 0 \quad \Rightarrow \quad \boxed{ A = 0 } %
    \end{equation*}
    La soluzione si riduce alla sola parte sinusoidale: $u(x) = B\sin(kx)$. %

    \item \textbf{Condizione in $\mathbf{x=a}$:} %
    \begin{equation*}
        u(a) = B\sin(ka) = 0 %
    \end{equation*}
    Scartiamo la soluzione banale $B=0$ (altrimenti non ci sarebbe nessuna particella). Affinché il seno si annulli, l'argomento deve essere un multiplo intero di $\pi$: %
    \begin{equation}
        ka = n\pi \quad \Rightarrow \quad \boxed{ k_n = \frac{n\pi}{a} } \quad \text{con } n = 1, 2, 3 \dots %
    \end{equation}
    \textit{($n=0$ si scarta perché darebbe $k=0 \Rightarrow u(x)=0$, annullando di nuovo la particella).}
\end{itemize}

\noindent
\textbf{Conseguenza FONDAMENTALE:} %
Confinando la particella, il vettore d'onda $k$ non può più assumere valori continui, ma è \textbf{QUANTIZZATO}. Sostituendo $k_n$ nell'espressione dell'energia, otteniamo uno \textbf{Spettro Discreto}: %
\begin{equation}
    \boxed{ E_n = \frac{\hbar^2 k_n^2}{2m} = \frac{\hbar^2 \pi^2 n^2}{2 m a^2} } \quad \text{(Autovalori dell'Energia)} %
\end{equation}

\subsection{Normalizzazione e Autofunzioni} %

Le autofunzioni spaziali, dipendenti dal \textbf{numero quantico} $n$, sono: %
\begin{equation*}
    u_n(x) = B\sin(k_n x) = B\sin\left(\frac{n\pi x}{a}\right) %
\end{equation*}

\noindent
Troviamo l'ampiezza $B$ imponendo la condizione di normalizzazione totale sulla probabilità: %
\begin{equation*}
    \int_0^a |\Psi|^2 \, dx = 1 \quad \Rightarrow \quad \int_0^a |B|^2 \sin^2\left(\frac{n\pi x}{a}\right) \, dx = 1 %
\end{equation*}
L'integrale del seno al quadrato su un numero intero di semiperiodi restituisce metà dell'intervallo ($a/2$). Quindi: %
\begin{equation*}
    |B|^2 \frac{a}{2} = 1 \quad \Rightarrow \quad \boxed{ B = \sqrt{\frac{2}{a}} } %
\end{equation*}

\begin{quote}
    \textbf{Soluzione Completa (Autovalori e Autofunzioni di $\hat{H}$):} %
    \begin{align}
        E_n &= \frac{\hbar^2 \pi^2}{2 m a^2} n^2 \\ %
        u_n(x) &= \sqrt{\frac{2}{a}} \sin\left(\frac{n\pi x}{a}\right) \quad \text{per } 0 \le x \le a %
    \end{align}
\end{quote}

    \begin{minipage}{0.7\textwidth}
        \textbf{Osservazioni Fisiche:} %
        \begin{itemize}
            \item I livelli di energia crescono con il quadrato del numero quantico ($E_n = n^2 E_1$). I livelli si distanziano sempre di più man mano che si sale. %
            \item Il livello fondamentale ($n=1$) \textbf{NON ha energia nulla} ($E_1 > 0$). Questo si chiama \textit{Energia di Punto Zero} ed è una conseguenza diretta del Principio di Indeterminazione di Heisenberg (se la particella fosse ferma con $E=0$, avrebbe $\Delta p = 0$, ma essendo confinata in $a$ ha un $\Delta x$ finito, il che violerebbe il principio). %
            \item Il modulo quadro di queste funzioni $|u_n(x)|^2$ fornisce la densità di probabilità di trovare la particella. Nello stato $n=2$, ad esempio, la probabilità di trovare la particella esattamente al centro della buca ($x=a/2$) è rigorosamente \textbf{ZERO} (nodo della funzione d'onda)! %
        \end{itemize}
    \end{minipage}
\end{figure}

\subsection{Proprietà Aggiuntive (Buca Infinita 1D)} %

\noindent
\textbf{1. Assenza dello stato $n=0$:} %
Il livello $n=0$ NON È PERMESSO. Se $n=0 \Rightarrow E=0$. La particella sarebbe ferma ($\Delta p = 0$). Tuttavia, l'incertezza sulla posizione $\Delta x$ vale al massimo $a$ (la larghezza della buca). Di conseguenza: %
\begin{equation*}
    \Delta x \Delta p = a \cdot 0 = 0 %
\end{equation*}
Questo violerebbe il Principio di Heisenberg. Deve quindi esistere un'Energia di punto zero $E_1 = \frac{\hbar^2 \pi^2}{2ma^2}$ %.

\noindent
\textbf{2. Parità delle Autofunzioni:} %
Rispetto al centro della buca ($x = a/2$), le funzioni d'onda hanno una PARITÀ ben definita: %
\begin{itemize}
    \item $n=1 \Rightarrow$ PARI (rispetto ad $a/2$) %
    \item $n=2 \Rightarrow$ DISPARI %
    \item $n=3 \Rightarrow$ PARI %
\end{itemize}
Regola aurea: $n$ dispari $\Rightarrow$ pari; $n$ pari $\Rightarrow$ dispari \textit{(Nota: se avessimo centrato la buca tra $-a/2$ e $a/2$, avremmo ottenuto direttamente seni per i dispari e coseni per i pari)} %.

\noindent
\textbf{3. Nodi e Ortogonalità:} %
Se il numero quantico $n$ aumenta, aumenta anche il numero di nodi (punti in cui $\Psi=0$) all'interno della buca.

\begin{enumerate}
    \setcounter{enumi}{3}
    \item \textbf{Ortogonalità:} Le autofunzioni formano una base ortonormale, verificabile tramite l'integrale del loro prodotto:
    \begin{equation}
        \int_0^a \sqrt{\frac{2}{a}} \sin\left(\frac{n\pi x}{a}\right) \cdot \sqrt{\frac{2}{a}} \sin\left(\frac{m\pi x}{a}\right) dx = \delta_{n,m} \quad \Rightarrow \quad \text{ortogonali}
    \end{equation}

    \item \textbf{Stato Generico e Probabilità:} Una qualsiasi funzione d'onda $\Psi(x)$ nella buca può essere scritta come serie delle autofunzioni:
    \begin{equation*}
        \Psi(x) = \sum_n c_n u_n(x) = \sqrt{\frac{2}{a}} \sum_n c_n \sin\left(\frac{n\pi x}{a}\right)
    \end{equation*}
    Calcoliamo i coefficienti $c_n$ proiettando la funzione $\Psi$ sull'autostato ennesimo:
    \begin{equation*}
        c_n = \int_0^a u_n^*(x) \Psi(x) \, dx = \sqrt{\frac{2}{a}} \int_0^a \sin\left(\frac{n\pi x}{a}\right) \Psi(x) \, dx
    \end{equation*}
    Per il postulato della misura, il modulo quadro del coefficiente è la probabilità di misurare quell'energia, e la loro somma deve fare $1$:
    \begin{equation*}
        |c_n|^2 = \mathbb{P}(E = E_n) \quad \Rightarrow \quad \sum_n |c_n|^2 = 1
    \end{equation*}

    \item \textbf{Cambio degli Estremi:} Se CAMBIO gli ESTREMI della buca (es. spostando l'origine al centro), la forma matematica delle $u_n(x)$ cambia.
    Le nuove condizioni al contorno impongono che la funzione si annulli sui nuovi bordi:
    \begin{equation*}
        u\left(x = \pm \frac{a}{2}\right) = 0
    \end{equation*}
    \textit{(Questo porterà ad avere seni per gli stati dispari e coseni per gli stati pari).}

    \item \textbf{Evoluzione Temporale Completa:} Unendo la parte spaziale e il fattore di fase temporale, la soluzione generale tempo-dipendente è:
    \begin{equation}
        \Psi(x,t) = \sum_n c_n \sqrt{\frac{2}{a}} \sin\left(\frac{n\pi x}{a}\right) e^{-i\frac{E_n}{\hbar}t}
    \end{equation}

    \item \textbf{Valore Medio del Momento:}
    \begin{equation}
        \langle \hat{p} \rangle = 0 \quad \Rightarrow \quad \text{Autostati REALI, in media la } v \text{ è } 0
    \end{equation}
    \textit{(Essendo la particella confinata, rimbalza continuamente a destra e sinistra, annullando la velocità vettoriale media).}
\end{enumerate}

\noindent
\textbf{9. Raccordo con il Limite Classico} %

\noindent
Classicamente, la probabilità di trovare una particella che rimbalza in una scatola è inversamente proporzionale alla sua velocità. Il tempo $dt$ speso tra $x$ e $x+dx$ è $dt = \frac{dx}{v}$: %
\begin{equation*}
    P_{class}(x \div x+dx) \propto dt \quad \Rightarrow \quad P_{class} \propto \frac{1}{v} %
\end{equation*}
Essendo l'energia cinetica $E_k$ fissata (conservazione dell'energia in una buca ideale), $v$ è costante, e quindi la probabilità classica è uniforme in tutta la buca: $P_{class} = \frac{1}{a} = \text{costante}$ %.

\textbf{Macroscopico vs Microscopico:} %
Sia un corpuscolo macroscopico con $m = 10^{-14}$ kg e $v = 1$ mm/s in una scatola di $a = 10$ cm. La sua energia cinetica è $\sim 10^{-21}$ J %.
Calcolando il suo numero quantico da $E_n = \frac{\hbar^2 \pi^2 n^2}{2ma^2}$, troviamo che il sistema si trova in un numero quantico $n \simeq 10^{15}$ %. 

\noindent
\textbf{Spaziatura dei Livelli:} %
\begin{equation*}
    \delta E = E_{n+1} - E_n \simeq \frac{\hbar^2 \pi^2}{2ma^2} [(n+1)^2 - n^2] \simeq \frac{\hbar^2 \pi^2}{2ma^2} 2n \quad \Rightarrow \quad \delta E \propto n %
\end{equation*}
Tuttavia, l'errore relativo $\frac{\delta E}{E} \propto \frac{n}{n^2} = \frac{1}{n}$ %.
Per $n \simeq 10^{15}$, l'errore relativo tende a zero. Lo spettro diventa per tutti gli scopi pratici \textbf{CONTINUO}, permettendoci di definire una "Densità degli Stati" $D(k)$: %

\section{Particella Confinata su un Anello (Rotore Rigido 1D)} %

\begin{figure}[H]

        L'equazione di Schrödinger lungo la coordinata $x$ (srotolata sull'anello) è formalmente quella della particella libera: %
        \begin{equation*}
            -\frac{\hbar^2}{2m} \frac{d^2 u}{dx^2} = Eu \quad \Rightarrow \quad u(x) = A e^{ikx} %

A differenza della buca, qui non ci sono pareti dove la funzione si annulla. Tuttavia, \textbf{il punto $x$ e il punto $x+a$ coincidono fisicamente} %.
Imponiamo le \textbf{Condizioni al Contorno CICLICHE} (o di periodicità): %
\begin{equation*}
    u(x+a) = u(x) %
\end{equation*}
Sostituendo l'esponenziale: %
\begin{equation*}
    A e^{ik(x+a)} = A e^{ikx} \quad \Rightarrow \quad e^{ikx} e^{ika} = e^{ikx} \quad \Rightarrow \quad e^{ika} = 1 %
\end{equation*}
Affinché $e^{ika} = 1$, l'argomento deve essere un multiplo intero di $2\pi$: %
\begin{equation}
    k a = 2n\pi \quad \Rightarrow \quad \boxed{ k_n = \frac{2\pi n}{a} } \quad \text{con } n = 0, \pm 1, \pm 2, \dots %
\end{equation}
(La quantizzazione nasce dalla topologia dello spazio!)

L'energia per la particella sull'anello è quindi quantizzata: %
\begin{equation}
    \boxed{ E_n = \frac{\hbar^2 k_n^2}{2m} = \frac{4 \hbar^2 \pi^2 n^2}{2 m a^2} = \frac{2 \hbar^2 \pi^2}{m a^2} n^2 } %
\end{equation}

Troviamo la costante di normalizzazione $A$: %
\begin{equation*}
    \int_0^a |A e^{ikx}|^2 dx = 1 \quad \Rightarrow \quad |A|^2 a = 1 \quad \Rightarrow \quad A = \frac{1}{\sqrt{a}} %
\end{equation*}

\noindent
\textbf{Spettro Energetico e Degenerazione:} %
\begin{itemize}
    \item $n = 0 \Rightarrow u_0(x) = \frac{1}{\sqrt{a}}$ e $E_0 = 0$. Lo stato fondamentale ha energia nulla ed è \textbf{NON degenere} (il principio di Heisenberg qui è salvo perché se $\Delta p = 0$, la posizione è totalmente incerta su tutto l'anello, $\Delta x \to \infty$ virtualmente sul cerchio) %.
    \item $n = \pm 1 \Rightarrow E_1 = \frac{2 \hbar^2 \pi^2}{m a^2}$. Le funzioni d'onda sono $u_1 = \frac{1}{\sqrt{a}} e^{i\frac{2\pi}{a}x}$ e $u_{-1} = \frac{1}{\sqrt{a}} e^{-i\frac{2\pi}{a}x}$. Corrispondono a rotazioni in sensi opposti (momento angolare destro e sinistro) e condividono la stessa energia $\Rightarrow$ Livello \textbf{2 volte DEGENERE} %.
\end{itemize}

\end{equation*}

Abbiamo ben \textbf{5 incognite}: i quattro coefficienti $B, C, D, F$ e gli autovalori dell'energia $E_n$ (nascosti in $\alpha$ e $k$) %.
Per risolverlo, imponiamo le condizioni di continuità per la funzione d'onda $u(x)$ e per la sua derivata prima $u'(x)$ sui due bordi ($x=0$ e $x=a$): %

\begin{align*}
    1) \quad u_2(0^+) &= u_1(0^-) &\Rightarrow \quad & C + D = B \\ %

    4) \quad \left. \frac{du_3}{dx} \right|_{a^+} &= \left. \frac{du_2}{dx} \right|_{a^-} &\Rightarrow \quad & -\alpha F e^{-\alpha a} = i k C e^{ika} - i k D e^{-ika} %
\end{align*}

La quinta equazione è fornita dalla \textbf{Condizione di Normalizzazione} globale: %

\end{equation*}

\subsection{Caso 1: $\mathbf{E < V_0}$ (Riflessione Totale Quantistica)} %

Scriviamo l'equazione di Schrödinger nelle due regioni. %
\textbf{Regione 1 ($\mathbf{x < 0}$, $\mathbf{V=0}$):} %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{d^2 u}{dx^2} = E u \quad \Rightarrow \quad K^2 = \frac{2mE}{\hbar^2} \quad \Rightarrow \quad u_1(x) = A e^{iKx} + B e^{-iKx} %
\end{equation*}

\noindent
\textbf{Regione 2 ($\mathbf{x > 0}$, $\mathbf{V=V_0}$):} %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{d^2 u}{dx^2} + V_0 u = E u \quad \Rightarrow \quad \frac{d^2 u}{dx^2} = \underbrace{\frac{2m(V_0 - E)}{\hbar^2}}_{\alpha^2} u \quad \Rightarrow \quad u_2(x) = C e^{\alpha x} + D e^{-\alpha x} %
\end{equation*}
\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-04-23-17-44-28.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        Poiché per $x \to +\infty$ la funzione deve rimanere finita, scartiamo l'esponenziale crescente imponendo $C=0$. Rimane l'\textbf{Onda Evanescente}: $u_2(x) = D e^{-\alpha x}$ %.
        \textit{(Significato: ho una probabilità non nulla di trovare la particella per $x>0$, anche se l'energia è minore del gradino!)} %
    \end{minipage}
\end{figure}

\noindent
\textbf{Condizioni di Raccordo in $\mathbf{x=0}$:} %
\begin{align*}
    u_1(0) &= u_2(0) \quad \Rightarrow \quad A + B = D \\ %
    u'_1(0) &= u'_2(0) \quad \Rightarrow \quad iKA - iKB = -\alpha D %
\end{align*}
\textit{(Nota: le onde piane si estendono all'infinito, quindi $\int |u|^2 dx \to \infty$. Non sono normalizzabili al valore 1, ma possiamo ragionare sui flussi di probabilità relativi)}. %

\noindent
Sostituiamo $D$ nella seconda equazione: %
\begin{align*}
    iKA - iKB &= -\alpha(A + B) \\ %
    (iK + \alpha)A &= (iK - \alpha)B \quad \Rightarrow \quad \frac{B}{A} = \frac{iK + \alpha}{iK - \alpha} %
\end{align*}

\noindent
Calcoliamo il \textbf{Coefficiente di Riflessione $R$} (rapporto tra probabilità dell'onda riflessa $B$ e onda incidente $A$): %
\begin{equation}
    \boxed{ R = \frac{|B|^2}{|A|^2} = \frac{|iK + \alpha|^2}{|iK - \alpha|^2} = \frac{K^2 + \alpha^2}{K^2 + \alpha^2} = 1 } %
\end{equation}
Anche se la particella penetra un po' nella barriera, il flusso totale viene \textbf{interamente riflesso} ($R=1$). %

\noindent
Verifichiamo i flussi calcolando la densità di corrente $J = \frac{\hbar}{2mi} \left( \Psi^* \frac{\partial \Psi}{\partial x} - \Psi \frac{\partial \Psi^*}{\partial x} \right)$ per l'onda incidente $\Psi(x,t) = A e^{i(Kx - \omega t)}$: %
\begin{align*}
    J &= \frac{\hbar}{2mi} \left[ A^* e^{-i(Kx - \omega t)} (A i K) e^{i(Kx - \omega t)} - A e^{i(Kx - \omega t)} (-A^* i K) e^{-i(Kx - \omega t)} \right] = \\ %
      &= \frac{\hbar}{2mi} \left( i K |A|^2 + i K |A|^2 \right) = \frac{\hbar}{2mi} (2 i K |A|^2) = \frac{\hbar K}{m} |A|^2 = v |A|^2 %
\end{align*}

Il coefficiente di riflessione come rapporto tra flussi è: %
\begin{equation*}
    R = \frac{J_{\leftarrow}}{J_{\rightarrow}} = \frac{v |B|^2}{v |A|^2} = \frac{|B|^2}{|A|^2} = 1 %
\end{equation*}
Inoltre, nella Regione 2 la funzione $u_2(x)$ è un esponenziale reale, quindi la sua derivata è reale. Nella formula di $J$, $\Psi^* \Psi'$ e $\Psi (\Psi')^*$ si elidono esattamente, restituendo $J_2 = 0$. Tutta la probabilità che entra nella barriera, torna indietro! %

\subsection{Caso 2: $\mathbf{E > V_0}$ (Scattering e Trasmissione)} %

Se l'energia è superiore alla barriera, la particella può propagarsi in entrambe le regioni. %
\begin{align*}
    u_1(x) &= A e^{ikx} + B e^{-ikx} \quad &\text{con } k = \sqrt{\frac{2mE}{\hbar^2}} \\ %
    u_2(x) &= D e^{ik'x} \quad &\text{con } k' = \sqrt{\frac{2m(E - V_0)}{\hbar^2}} %
\end{align*}
\textit{(Assumiamo che non ci siano onde riflesse provenienti da $+\infty$, da cui il termine con $e^{-ik'x}$ nullo).} %

Imponendo i raccordi (che formalmente sono identici al Caso 1, sostituendo $\alpha$ con $-ik'$): %
\begin{equation*}
    \frac{B}{A} = \frac{k - k'}{k + k'} \quad \Rightarrow \quad R = \frac{|B|^2}{|A|^2} = \left| \frac{k - k'}{k + k'} \right|^2 %
\end{equation*}

Poiché $k \neq k'$, avremo sempre $R > 0$. %
\begin{quote}
    \textbf{Paradosso Quantistico:} Classicamente, se $E > V_0$, la particella passa il gradino col 100\% di probabilità (rallentando e basta). In Meccanica Quantistica, \textbf{c'è sempre una probabilità non nulla di essere riflessi dal gradino!} %
\end{quote}

Definiamo il \textbf{Coefficiente di Trasmissione $T$}: %
\begin{equation}

\end{equation}

\section{Oscillatore Armonico Quantistico} %

\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-04-24-13-51-48.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        L'oscillatore armonico è il modello di riferimento per qualsiasi sistema che compie piccole oscillazioni attorno a un punto di equilibrio stabile. Il potenziale è parabolico: $V(x) = \frac{1}{2}m\omega^2 x^2$. %
        L'Hamiltoniana del sistema è: %
        \begin{equation}
            \hat{H} = -\frac{\hbar^2}{2m} \frac{d^2}{dx^2} + \frac{1}{2}m\omega^2 x^2 %
        \end{equation}
    \end{minipage}
\end{figure}

Scriviamo l'equazione di Schrödinger indipendente dal tempo: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{d^2 u}{dx^2} + \frac{1}{2}m\omega^2 x^2 u = E u %
\end{equation*}

\noindent
Per semplificare l'equazione, effettuiamo un cambio di variabili. Dividiamo tutto per $\hbar\omega$ e introduciamo le variabili adimensionali: %
\begin{equation*}
    y = \sqrt{\frac{m\omega}{\hbar}} x \quad \Rightarrow \quad dy = \sqrt{\frac{m\omega}{\hbar}} dx \quad ; \quad \varepsilon = \frac{E}{\hbar\omega} %
\end{equation*}
Sostituendo e semplificando, l'equazione differenziale assume una forma compatta: %
\begin{equation}
    \frac{1}{2} \frac{d^2 u}{dy^2} = \left(\frac{1}{2} y^2 - \varepsilon\right) u \quad \Rightarrow \quad \frac{d^2 u}{dy^2} = (y^2 - 2\varepsilon) u %
\end{equation}

\textbf{Soluzione Asintotica ($y \to \pm\infty$):} %
Per $y \gg \varepsilon$, l'equazione diventa $\frac{d^2 u}{dy^2} \simeq y^2 u$. La soluzione asintotica fisicamente accettabile (che non diverge) è la Gaussiana: %
\begin{equation*}
    u_\infty(y) = e^{-y^2/2} %
\end{equation*}
Ipotizziamo quindi una soluzione generale del tipo: $u(y) = h(y) e^{-y^2/2}$ %. 
Sostituendo questa forma nell'equazione differenziale, si ottiene l'\textbf{Equazione differenziale di Hermite}: %
\begin{equation}
    \boxed{ \frac{d^2 h}{dy^2} - 2y \frac{dh}{dy} + (2\varepsilon - 1)h = 0 } %
\end{equation}

\subsection{Metodo dello Sviluppo in Serie di Potenze} %

Risolviamo l'equazione di Hermite esprimendo $h(y)$ come una serie infinita: %
\begin{equation*}
    h(y) = \sum_{k=0}^\infty a_k y^k %
\end{equation*}
Calcoliamo le derivate: %
\begin{align*}
    \frac{dh}{dy} &= \sum_{k=1}^\infty k a_k y^{k-1} \\ %
    \frac{d^2 h}{dy^2} &= \sum_{k=2}^\infty k(k-1) a_k y^{k-2} \xrightarrow{m=k-2} \sum_{m=0}^\infty (m+2)(m+1) a_{m+2} y^m %
\end{align*}

Sostituiamo nell'equazione differenziale (raccogliendo tutto sotto la stessa sommatoria $\sum y^m$): %
\begin{equation*}
    \sum_{m=0}^\infty \left[ (m+2)(m+1)a_{m+2} - 2m a_m + (2\varepsilon - 1)a_m \right] y^m = 0 %
\end{equation*}
Affinché sia zero per ogni $y$, i coefficienti devono essere nulli. Otteniamo la \textbf{Formula Ricorsiva}: %
\begin{equation}
    \boxed{ a_{m+2} = \frac{2m + 1 - 2\varepsilon}{(m+2)(m+1)} a_m } %
\end{equation}

\noindent
\textbf{Quantizzazione dell'Energia:} %
Se la serie non si troncasse, per $m \to \infty$ si comporterebbe come $e^{y^2}$, e moltiplicata per la gaussiana asintotica darebbe $e^{y^2} e^{-y^2/2} = e^{y^2/2}$, che diverge all'infinito ($\notin \mathcal{L}^2$) %.
Per evitare che la funzione d'onda esploda, la serie \textbf{DEVE TRONCARSI}. Esisterà un $m_{MAX} = n$ per cui $a_{n+2} = 0$. Imponiamo quindi che il numeratore si annulli: %
\begin{equation*}
    2n + 1 - 2\varepsilon = 0 \quad \Rightarrow \quad \varepsilon = n + \frac{1}{2} %
\end{equation*}
Ricordando che $\varepsilon = E / \hbar\omega$, otteniamo lo Spettro degli Autovalori dell'oscillatore armonico: %
\begin{equation}
    \boxed{ E_n = \left(n + \frac{1}{2}\right) \hbar\omega } \quad \text{con } n = 0, 1, 2, \dots \quad \text{(come Planck apparte per $\frac{1}{2}$ )} %
\end{equation}
\textit{Conclusione profonda: La quantizzazione dell'energia deriva direttamente dalla necessità di poter NORMALIZZARE la funzione d'onda!} %

\subsection{Polinomi di Hermite e Funzioni d'Onda} %

I polinomi generati dal troncamento della serie si chiamano \textbf{Polinomi di Hermite} $H_n(y)$. %
\begin{itemize}
    \item \textbf{Stato Fondamentale ($\mathbf{n=0}$):} $E_0 = \frac{\hbar\omega}{2}$. Essendo $a_2 = 0$, rimane solo una costante $h(y) = a_0$. %
    \begin{equation*}
        u_0(y) = a_0 e^{-y^2/2} \quad \text{(Curva Gaussiana)} %
    \end{equation*}
    \item \textbf{Primo Stato Eccitato ($\mathbf{n=1}$):} $E_1 = \frac{3}{2}\hbar\omega$. Si pone $a_0 = 0$ e $a_3 = 0$. Rimane $h(y) = a_1 y$. %
    \begin{equation*}
        u_1(y) = a_1 y e^{-y^2/2} \quad \text{(Gaussiana per retta)} %
    \end{equation*}
\end{itemize}

Aggiungendo la costante di normalizzazione, le autofunzioni esatte sono: %
\begin{equation}
    \boxed{ u_n(x) = \left(\frac{m\omega}{\pi\hbar}\right)^{1/4} \frac{1}{\sqrt{2^n n!}} H_n\left(\sqrt{\frac{m\omega}{\hbar}}x\right) e^{-\frac{m\omega x^2}{2\hbar}} } %
\end{equation}

\section{Problemi Quantistici in Campo Centrale} %

Studiamo ora i sistemi in cui l'energia potenziale dipende solo dalla distanza dall'origine (forze centrali), ovvero $V = V(r)$. È il caso dell'atomo di idrogeno (attrazione elettrostatica nucleo-elettrone). %

Conviene passare dalle coordinate cartesiane $(x,y,z)$ a quelle sferiche $(r, \theta, \varphi)$: %
\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-04-24-14-28-58.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        \begin{align*}
            x &= r \sin\theta \cos\varphi \\ %
            y &= r \sin\theta \sin\varphi \\ %
            z &= r \cos\theta %
        \end{align*}
    \end{minipage}
\end{figure}
Con i rispettivi domini e l'elemento di volume $d\tau$: %
\begin{equation*}
    r \in [0, +\infty] \quad , \quad \varphi \in [0, 2\pi] \quad , \quad \theta \in [0, \pi] \qquad \Rightarrow \quad d\tau = r^2 dr \sin\theta d\theta d\varphi %
\end{equation*}

\noindent
L'equazione di Schrödinger dipendente dal tempo $i\hbar \frac{\partial \Psi}{\partial t} = \hat{H}\Psi$ per gli stati stazionari $\Psi(x,y,z,t) = u(r,\theta,\varphi)f(t)$ diventa: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \Delta u(r,\theta,\varphi) + V(r) u(r,\theta,\varphi) = E u(r,\theta,\varphi) %
\end{equation*}

L'operatore Laplaciano $\Delta$ in coordinate sferiche è piuttosto complesso. Lo scomponiamo in una parte puramente radiale e in un operatore angolare $\hat{O}_{\theta,\varphi}$: %
\begin{equation}
    \Delta = \frac{1}{r^2} \frac{\partial}{\partial r} \left( r^2 \frac{\partial}{\partial r} \right) + \frac{1}{r^2} \underbrace{ \left[ \frac{1}{\sin\theta} \frac{\partial}{\partial \theta} \left( \sin\theta \frac{\partial}{\partial \theta} \right) + \frac{1}{\sin^2\theta} \frac{\partial^2}{\partial \varphi^2} \right] }_{\hat{O}_{\theta,\varphi}} %
\end{equation}
Sostituendolo nell'equazione: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \left[ \frac{1}{r^2} \frac{\partial}{\partial r} \left( r^2 \frac{\partial u}{\partial r} \right) + \frac{1}{r^2} \hat{O}_{\theta,\varphi} u \right] + V(r)u = E u %
\end{equation*}

\subsection{Prima Separazione delle Variabili (Radiale e Angolare)} %

Cerchiamo soluzioni del tipo: $u(r,\theta,\varphi) = R(r) Y(\theta,\varphi)$. %
Sostituiamo nell'equazione differenziale, moltiplichiamo per $-\frac{2mr^2}{\hbar^2}$ e dividiamo per $R Y$: %
\begin{equation*}
    \underbrace{ \frac{1}{R} \frac{d}{dr} \left( r^2 \frac{dR}{dr} \right) - \frac{2mr^2}{\hbar^2}(V - E) }_{\text{Dipende solo da } r} + \underbrace{ \frac{1}{Y} \hat{O}_{\theta,\varphi} Y }_{\text{Dipende solo da } \theta, \varphi} = 0 %
\end{equation*}

\noindent
Affinché la somma di due funzioni con variabili indipendenti sia sempre zero, entrambe devono essere uguali a una costante opposta. Scegliamo opportunamente la costante di separazione $l(l+1)$: %

\begin{enumerate}
    \item \textbf{Parte Radiale:} %
    \begin{equation}
        \frac{1}{R} \frac{d}{dr} \left( r^2 \frac{dR}{dr} \right) - \frac{2mr^2}{\hbar^2}(V - E) = l(l+1) %
    \end{equation}
    \item \textbf{Parte Angolare:} %
    \begin{equation}
        \frac{1}{Y} \left[ \frac{1}{\sin\theta} \frac{\partial}{\partial \theta} \left( \sin\theta \frac{\partial Y}{\partial \theta} \right) + \frac{1}{\sin^2\theta} \frac{\partial^2 Y}{\partial \varphi^2} \right] = -l(l+1) %
    \end{equation}
    $\hookrightarrow$ \textit{La parte angolare non richiede l'espressione specifica del potenziale $V(r)$, è universale per tutti i campi centrali!} %
\end{enumerate}

\textit{Nota sulla normalizzazione:} $\int |u|^2 r^2 dr \sin\theta d\theta d\varphi = \int_0^{+\infty} |R|^2 r^2 dr \int_0^{2\pi} \int_0^\pi |Y|^2 \sin\theta d\theta d\varphi = 1$ %.

\subsection{Seconda Separazione delle Variabili (Armoniche Sferiche)} %

Risolviamo l'equazione angolare separando a sua volta $Y(\theta,\varphi) = F(\theta)G(\varphi)$ %.
Sostituendo nell'equazione della parte angolare, dividendo per $FG$ e moltiplicando tutto per $\sin^2\theta$: %
\begin{equation*}
    \underbrace{ \frac{\sin\theta}{F} \frac{d}{d\theta} \left( \sin\theta \frac{dF}{d\theta} \right) + \sin^2\theta \cdot l(l+1) }_{\text{COST } m^2} = \underbrace{ -\frac{1}{G} \frac{d^2 G}{d\varphi^2} }_{\text{COST } m^2} %
\end{equation*}
Abbiamo isolato $\theta$ da $\varphi$ uguagliandole a una nuova costante $m^2$ %.

\noindent
\textbf{Soluzione per $\mathbf{\varphi}$ (Equazione Azimutale):} %
\begin{equation*}
    -\frac{1}{G} \frac{d^2 G}{d\varphi^2} = m^2 \quad \Rightarrow \quad \frac{d^2 G}{d\varphi^2} + m^2 G = 0 \quad \Rightarrow \quad G(\varphi) = e^{im\varphi} %
\end{equation*}

\noindent
\textbf{Condizione al contorno (Ciclicità):} Affinché la funzione d'onda sia a singolo valore (univoca in ogni punto dello spazio), deve coincidere dopo un giro completo ($2\pi$) %:
\begin{equation*}
    G(\varphi + 2\pi) = G(\varphi) \quad \Rightarrow \quad e^{im(\varphi + 2\pi)} = e^{im\varphi} \quad \Rightarrow \quad e^{im2\pi} = 1 %
\end{equation*}
Questo è vero se e solo se $m$ è un intero: %
\begin{equation}
    \boxed{ m = 0, \pm 1, \pm 2, \pm 3 \dots } \quad \text{(\textbf{NUMERO QUANTICO MAGNETICO})} %
\end{equation}

\subsection{Polinomi di Legendre (Equazione per $\mathbf{\theta}$)} %

L'equazione rimanente per $F(\theta)$ diventa: %
\begin{equation*}
    \frac{\sin\theta}{F} \frac{d}{d\theta} \left( \sin\theta \frac{dF}{d\theta} \right) + \sin^2\theta \cdot l(l+1) = m^2 %
\end{equation*}
La soluzione porta ai \textbf{Polinomi Associati di Legendre} $P_l^m(\cos\theta)$: %
\begin{equation}
    F(\theta) = P_l^m(\cos\theta) = (-1)^m (\sin\theta)^m \frac{d^m}{d(\cos\theta)^m} \underbrace{ \left[ \frac{1}{2^l l!} \frac{d^l}{d(\cos\theta)^l} (\cos^2\theta - 1)^l \right] }_{\text{Polinomi di Legendre}} %
\end{equation}

\noindent
Affinché queste funzioni "vadano bene in fisica" e \textbf{NON DIVERGANO} (così da poter essere normalizzate), devono essere rispettate delle ferree condizioni sui parametri: %
\begin{align*}
    1) \quad & l \ge 0 \text{ (intero positivo)} \\ %
    2) \quad & l \ge |m| %
\end{align*}

Questa limitazione matematica impone la quantizzazione del secondo parametro: %
\begin{align}
    \boxed{ l = 0, 1, 2, \dots } \quad &\text{(\textbf{NUMERO QUANTICO ORBITALE})} \label{eq:orb} \\ %
    m \in [-l, +l] \quad &\Rightarrow \quad m = -l, -l+1, \dots , 0, \dots , l-1, l \label{eq:mag} %
\end{align}

\section{Armoniche Sferiche} %

Riunendo la parte colatitudinale e azimutale, definiamo le \textbf{Armoniche Sferiche} $Y_l^m(\theta, \varphi)$: %
\begin{equation}
    \boxed{ Y_l^m(\theta, \varphi) = \text{cost} \cdot P_l^m(\cos\theta) e^{im\varphi} } %
\end{equation}

I numeri quantici che definiscono questi stati angolari sono: %
\begin{align*}
    l &\to \text{Numero quantico ORBITALE} = 0, 1, 2, 3, \dots \\ %
    m &\to \text{Numero quantico MAGNETICO} = -l, \dots, +l %
\end{align*}

\noindent
Ad ogni valore di $l$ è associata una lettera spettroscopica storica, e il numero di stati magnetici possibili è sempre $2l + 1$: %
\begin{itemize}
    \item $l = 0 \quad ; \quad m = 0 \quad \Rightarrow \quad$ \textbf{STATI s} (1 stato) %
    \item $l = 1 \quad ; \quad m = 0, \pm 1 \quad \Rightarrow \quad$ \textbf{STATI p} (3 stati) %
    \item $l = 2 \quad ; \quad m = 0, \pm 1, \pm 2 \quad \Rightarrow \quad$ \textbf{STATI d} (5 stati) %
    \item $l = 3 \quad ; \quad m = 0, \pm 1, \pm 2, \pm 3 \quad \Rightarrow \quad$ \textbf{STATI f} (7 stati) %
\end{itemize}

\textit{Esempi delle prime armoniche sferiche:} %
\begin{equation*}
    Y_0^0 = \frac{1}{\sqrt{4\pi}} \quad , \quad Y_1^0 = \sqrt{\frac{3}{4\pi}}\cos\theta \quad , \quad Y_1^{\pm 1} = \mp \sqrt{\frac{3}{8\pi}}\sin\theta e^{\pm i\varphi} %
\end{equation*}

\subsection{Proprietà delle Armoniche Sferiche} %

\begin{enumerate}
    \item \textbf{Ortonormalità:} %
    \begin{equation*}
        \int_0^{2\pi} \int_0^\pi \left[ Y_l^m \right]^* Y_{l'}^{m'} \sin\theta d\theta d\varphi = \delta_{ll'} \delta_{mm'} %
    \end{equation*}
    
    \item \textbf{Insieme Completo:} Qualsiasi funzione angolare può essere scritta come loro combinazione lineare: %
    \begin{equation*}
        f(\theta, \varphi) = \sum_{l,m} c_{lm} Y_l^m(\theta, \varphi) %
    \end{equation*}
    
    \item \textbf{Inversione delle Coordinate (Parità):} %
    Se inverto il vettore posizione $\vec{r} \to -\vec{r}$, le coordinate sferiche cambiano così: %
    \begin{equation*}
        r \to r \quad , \quad \theta \to \pi - \theta \quad , \quad \varphi \to \varphi + \pi %
    \end{equation*}
    Applicando questa trasformazione all'armonica sferica: %
    \begin{equation*}
        Y_l^m(\pi - \theta, \varphi + \pi) = (-1)^l Y_l^m(\theta, \varphi) %
    \end{equation*}
    $\Rightarrow$ Se $l$ è PARI, $Y_l^m$ non cambia segno. Se $l$ è DISPARI, $Y_l^m$ cambia segno! %
    
    \item \textbf{Densità di Probabilità Angolare:} %
    \begin{equation*}
        |Y_l^m|^2 \sin\theta d\theta d\varphi = |Y_l^m|^2 d\Omega \quad \text{(Prob. di trovare la particella in } d\Omega\text{)} %
    \end{equation*}
    Poiché il modulo quadro cancella la parte complessa $e^{im\varphi} \cdot e^{-im\varphi} = 1$, la probabilità angolare \textbf{dipende solo da $\theta$}! (Simmetria cilindrica attorno all'asse z). %
\end{enumerate}

\section{Momento Angolare} %

In un campo centrale $V(r)$, la simmetria sferica implica la conservazione non solo dell'Energia, ma anche del \textbf{Momento Angolare} %. 

Richiamiamo l'operatore momento angolare quantistico: %
\begin{equation*}
    \hat{\vec{L}} = \hat{\vec{r}} \wedge \hat{\vec{p}} = \hat{\vec{r}} \wedge (-i\hbar\nabla) %
\end{equation*}

\noindent
\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-04-24-14-52-40.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        Passando alle coordinate sferiche $(r, \theta, \varphi)$, calcoliamo esplicitamente la componente lungo l'asse z: %
    \end{minipage}
\end{figure}
\begin{equation*}
    \hat{L}_z = x\hat{p}_y - y\hat{p}_x = -i\hbar \left( x \frac{\partial}{\partial y} - y \frac{\partial}{\partial x} \right) \quad \xrightarrow{\text{coord. sferiche}} \quad \boxed{ \hat{L}_z = -i\hbar \frac{\partial}{\partial \varphi} } %
\end{equation*}

Calcoliamo ora l'operatore \textbf{Momento Angolare al Quadrato} (modulo quadro): %
\begin{equation*}
    \hat{L}^2 = \hat{L}_x^2 + \hat{L}_y^2 + \hat{L}_z^2 = -\hbar^2 \underbrace{ \left[ \frac{1}{\sin\theta} \frac{\partial}{\partial \theta} \left( \sin\theta \frac{\partial}{\partial \theta} \right) + \frac{1}{\sin^2\theta} \frac{\partial^2}{\partial \varphi^2} \right] }_{\hat{O}_{\theta,\varphi}} %
\end{equation*}
\textit{Sorpresa straordinaria!} L'operatore angolare $\hat{O}_{\theta,\varphi}$ isolato durante la separazione delle variabili nell'equazione di Schrödinger non è altro che l'operatore $\hat{L}^2$ (a meno della costante $-\hbar^2$). Ovvero: $\hat{O}_{\theta,\varphi} = -\frac{1}{\hbar^2}\hat{L}^2$ %.

\subsection{Equazioni agli Autovalori} %

Sostituiamo questa scoperta nell'equazione differenziale angolare che definiva le armoniche sferiche $Y_l^m$: %
\begin{equation*}
    \frac{1}{Y_l^m} \hat{O}_{\theta,\varphi} Y_l^m = -l(l+1) \quad \Rightarrow \quad -\frac{1}{\hbar^2} \hat{L}^2 Y_l^m = -l(l+1) Y_l^m %
\end{equation*}
\begin{equation}
    \boxed{ \hat{L}^2 Y_l^m(\theta, \varphi) = \hbar^2 l(l+1) Y_l^m(\theta, \varphi) } %
\end{equation}
Le armoniche sferiche sono quindi \textbf{AUTOFUNZIONI dell'operatore $\hat{L}^2$}, con autovalori $\hbar^2 l(l+1)$. Poiché $l = 0, 1, 2 \dots$, il modulo del momento angolare è \textbf{SEMPRE QUANTIZZATO}! %

\noindent
Verifichiamo la stessa cosa per $\hat{L}_z$: %
\begin{equation*}
    \hat{L}_z Y_l^m(\theta, \varphi) = -i\hbar \frac{\partial}{\partial \varphi} \left[ P_l^m(\cos\theta) e^{im\varphi} \right] = -i\hbar (im) P_l^m e^{im\varphi} = m\hbar Y_l^m %
\end{equation*}
\begin{equation}
    \boxed{ \hat{L}_z Y_l^m(\theta, \varphi) = m\hbar Y_l^m(\theta, \varphi) } %
\end{equation}
Le $Y_l^m$ sono \textbf{SIMULTANEAMENTE AUTOFUNZIONI anche di $\hat{L}_z$}, con autovalori $m\hbar$. La proiezione lungo z è sempre quantizzata.\\

\noindent
\textit{Esempio:} Poniamo $l = 1$. Il modulo del vettore momento angolare è fissato: %
\begin{equation*}
    |\vec{L}| = \sqrt{\hbar^2 l(l+1)} = \sqrt{2}\hbar %
\end{equation*}

\subsection{Indeterminazione sulle Componenti} %

Abbiamo dimostrato che $\hat{L}^2$ e $\hat{L}_z$ condividono le stesse autofunzioni $Y_l^m$. Questo implica che commutano: %
\begin{equation*}
    [\hat{L}^2, \hat{L}_z] = 0 \quad (\text{e simmetricamente } [\hat{L}^2, \hat{L}_x] = 0 \, , \, [\hat{L}^2, \hat{L}_y] = 0) %
\end{equation*}
$\Rightarrow$ In un sistema quantistico posso conoscere \textbf{contemporaneamente e senza incertezza} sia il modulo $|\vec{L}|$ sia \textbf{UNA SOLA} delle sue proiezioni (tipicamente si sceglie $z$) %.

\noindent
Ma che succede alle altre componenti $L_x$ e $L_y$? Sapendo che $[\hat{L}_x, \hat{L}_y] = i\hbar\hat{L}_z$, applichiamo Heisenberg generalizzato: %
\begin{equation*}
    \sigma_{L_x} \sigma_{L_y} \ge \frac{1}{2} |\langle [\hat{L}_x, \hat{L}_y] \rangle| = \frac{1}{2} |\langle i\hbar \hat{L}_z \rangle| = \frac{\hbar}{2} |\langle \hat{L}_z \rangle| %
\end{equation*}
Se lo stato si trova perfettamente allineato sull'asse z (cioè $\vec{L}$ giace esattamente su z), allora $L_x$ e $L_y$ sarebbero rigorosamente uguali a $0$, e di conseguenza $\sigma_{L_x} = \sigma_{L_y} = 0$ %.
Ma se $\langle \hat{L}_z \rangle \neq 0$, allora $0 \ge \text{costante} > 0$, che è un \textbf{assurdo matematico}!\\

\noindent
\textbf{Conclusione Epocale:} Il vettore Momento Angolare $\vec{L}$ \textbf{NON PUÒ MAI ALLINEARSI PERFETTAMENTE} ad un asse! % 
La sua proiezione massima misurabile non è mai uguale al suo modulo (ad es. per $l=1$, $|\vec{L}| = \sqrt{2}\hbar$, ma $L_{z,max} = 1\hbar$). Se si allineasse, non avrei incertezza sulle altre componenti e violerei Heisenberg. \textbf{Questo vale per \textit{qualsiasi} vettore quantistico!} %

\section{Parte Radiale di $u(r,\theta,\varphi)$} %

Per un potenziale centrale $V(r) \rightarrow$ Simmetria sferica. %
Riscriviamo l'Hamiltoniana $\hat{H}$ sostituendo l'operatore angolare $\hat{O}_{\theta,\varphi} = -\frac{\hat{L}^2}{\hbar^2}$: %
\begin{align*}
    \hat{H} &= -\frac{\hbar^2}{2m} \left[ \frac{1}{r^2} \frac{\partial}{\partial r} \left( r^2 \frac{\partial}{\partial r} \right) + \frac{1}{r^2} \hat{O}_{\theta,\varphi} \right] + V(r) \\ %
    \hat{H} &= -\frac{\hbar^2}{2m} \frac{1}{r^2} \frac{\partial}{\partial r} \left( r^2 \frac{\partial}{\partial r} \right) + \underbrace{ \frac{\hat{L}^2}{2mr^2} }_{\text{PARTE ANGOLARE}} + V(r) %
\end{align*}

\noindent
Dall'equazione di Schrödinger $\hat{H}u(r,\theta,\varphi) = E u(r,\theta,\varphi)$, sappiamo che la soluzione si separa in $u(r,\theta,\varphi) = R(r)Y_l^m(\theta,\varphi)$ %.
Applichiamo l'operatore $\hat{L}^2$ all'intera funzione d'onda: %
\begin{align*}
    \underline{ \hat{L}^2 u(r,\theta,\varphi) } &= \hat{L}^2 [R(r) Y_l^m(\theta,\varphi)] = R(r) \hat{L}^2 Y_l^m(\theta,\varphi) = \\ %
    &= R(r) \hbar^2 l(l+1) Y_l^m(\theta,\varphi) = \hbar^2 l(l+1) R(r) Y_l^m(\theta,\varphi) = \\ %
    &= \underline{\underline{ \hbar^2 l(l+1) u(r,\theta,\varphi) }} %
\end{align*}

Poiché $\hat{H}$ e $\hat{L}^2$ condividono le stesse autofunzioni $u(r,\theta,\varphi)$, essi commutano: %
\begin{equation*}
    \Rightarrow \quad [\hat{H}, \hat{L}^2] = 0 \quad \text{($u(r,\theta,\varphi)$ in comune) $\Rightarrow \hat{L}$ si conserva quindi commuta con $\hat{H}$} %
\end{equation*}

Allo stesso modo, questo vale per la componente z del momento angolare: %
\begin{equation*}
    \Rightarrow \quad [\hat{H}, \hat{L}_z] = 0 \quad \text{con autostati le } u(r,\theta,\varphi) \text{ in \underline{SIMMETRIA SFERICA}} %
\end{equation*}

\subsection{I Tre Numeri Quantici (3D)} %
Essendo in 3 dimensioni (\textbf{3D}), il sistema è descritto da 3 numeri quantici associati a 3 operatori: %
\begin{itemize}
    \item $\hat{L}^2 u = \hbar^2 l(l+1) u \quad \Rightarrow \quad$ \textbf{($l$)} NUM. QUANT. ORBITALE %
    \item $\hat{L}_z u = \hbar m u \quad \Rightarrow \quad$ \textbf{($m$)} NUM. QUANT. MAGN. %
    \item $\hat{H} u = E u \quad \Rightarrow \quad$ ULTIMO NUM. QUANTICO \textbf{($n$)} $\Rightarrow$ NUM. QUANT. PRINCIPALE %
\end{itemize}

Infine, applicando la separazione, ricaviamo l'equazione per la sola parte radiale $R$: %
\begin{equation}
    \Rightarrow \quad\boxed{ \frac{1}{R} \frac{d}{dr} \left( r^2 \frac{dR}{dr} \right) - \frac{2mr^2}{\hbar^2}[V(r) - E] = l(l+1) \qquad (l = 0, 1, 2 \dots)}
\end{equation}

\subsection{Atomo di Idrogeno e Idrogenoidi} %

Per l'atomo di idrogeno (e ioni idrogenoidi come $He^+$, $Li^{2+}$), il potenziale è quello di Coulomb (attrazione elettrostatica tra un nucleo di carica $+Ze$ e un elettrone di carica $-e$): %
\begin{equation*}
    V(r) = -\frac{Z e^2}{4\pi\varepsilon_0 r} %
\end{equation*}

\noindent
Sostituiamo il potenziale nell'equazione radiale moltiplicata per $-\frac{\hbar^2}{2mr^2} R$: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{1}{r^2} \frac{d}{dr} \left( r^2 \frac{dR}{dr} \right) + \frac{\hbar^2 l(l+1)}{2mr^2} R - \frac{Z e^2}{4\pi\varepsilon_0 r} R = E R %
\end{equation*}

\noindent
\textbf{1. Riduzione dell'Equazione ($g(r) = r R(r)$):} %
Per semplificare la derivata seconda, effettuiamo la sostituzione $g(r) = r R(r)$. Svolgendo le derivate del prodotto, l'equazione assume una forma del tutto analoga a quella di Schrödinger unidimensionale: %
\begin{equation}
    \boxed{ -\frac{\hbar^2}{2m} \frac{d^2 g}{dr^2} + \left[ \frac{\hbar^2 l(l+1)}{2mr^2} - \frac{Z e^2}{4\pi\varepsilon_0 r} \right] g = E g } %
\end{equation}

Tutto il blocco tra parentesi quadre agisce come un \textbf{Potenziale Efficace ($V_{eff}$)}: %
\begin{equation}
    V_{eff} = \underbrace{ \frac{\hbar^2 l(l+1)}{2mr^2} }_{\text{Termine Centrifugo}} - \frac{Z e^2}{4\pi\varepsilon_0 r} = \underbrace{ \frac{L^2}{2mr^2} }_{\text{Termine Centrifugo}} - \frac{Z e^2}{4\pi\varepsilon_0 r} %
\end{equation}

\subsection{Analisi Asintotica} %

Prima di cercare la soluzione esatta (che richiede i polinomi di Laguerre), studiamo il comportamento della funzione $g(r)$ agli estremi del dominio. %

\noindent
\textbf{A. Limite per $\mathbf{r \to 0}$ (vicino al nucleo):} %
Per $r$ piccolissimo, il termine centrifugo in $1/r^2$ domina su tutti gli altri termini (incluso $E$ e $1/r$). L'equazione si approssima a: %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{d^2 g}{dr^2} = -\frac{\hbar^2 l(l+1)}{2mr^2} g \quad \Rightarrow \quad \frac{d^2 g}{dr^2} = \frac{l(l+1)}{r^2} g %
\end{equation*}
La cui soluzione compatibile (che non diverge negativamente in zero) è una potenza: %
\begin{equation}
    g(r) \propto r^{l+1} \quad \Rightarrow \quad R(r) \propto r^l %
\end{equation}

\noindent
\textbf{B. Limite per $\mathbf{r \to +\infty}$ (lontano dal nucleo):} %
A grande distanza, $V_{eff} \to 0$. Rimane solo il termine dell'energia totale. Poiché cerchiamo \textbf{Stati Legati}, sappiamo che l'energia deve essere negativa ($E < 0$). %
\begin{equation*}
    -\frac{\hbar^2}{2m} \frac{d^2 g}{dr^2} = E g \quad \Rightarrow \quad \frac{d^2 g}{dr^2} = -\underbrace{\frac{2mE}{\hbar^2}}_{> 0} g %
\end{equation*}
Definendo $K^2 = -\frac{2mE}{\hbar^2} > 0$, l'equazione diventa $\frac{d^2 g}{dr^2} = K^2 g$. %
La soluzione $e^{+Kr}$ diverge per $r \to +\infty$ e non va bene fisicamente. Dobbiamo tenere solo l'esponenziale decrescente: %
\begin{equation}
    g(r) \propto e^{-Kr} %
\end{equation}

\noindent
\textbf{C. La Soluzione Generale:} %
Unendo i due comportamenti asintotici, supponiamo che la soluzione esatta abbia la forma: %
\begin{equation}
    \boxed{ g(r) = \underbrace{r^{l+1}}_{r \to 0} \underbrace{e^{-Kr}}_{r \to +\infty} \underbrace{h(r)}_{\text{Da determinare}} } \qquad \qquad h(r \to 0) \to \text{COST}  %
\end{equation}

\section{Soluzione dell'Equazione Radiale (Idrogeno)} %

Ritorniamo all'equazione per la funzione $g(r) = r R(r)$: %
\begin{equation*}
    g(r) \propto r^{l+1} e^{-Kr} h(r) \quad \text{con } K = \sqrt{-\frac{2mE}{\hbar^2}} %
\end{equation*}

\noindent
Effettuiamo un cambio di variabile definendo $\rho = 2Kr$. L'equazione differenziale per $h(\rho)$ diventa: %
\begin{equation}
    \Rightarrow g(\rho) = \left( \frac{\rho}{2\kappa} \right)^{l+1} e^{-\frac{\rho}{2}} h(\rho), \qquad -\frac{\hbar^2}{2m} \frac{d^2 g}{dr^2} + \left[ \frac{\hbar^2 l(l+1)}{2mr^2} - \frac{Ze^2}{4\pi\varepsilon_0 r} \right] g = Eg
\end{equation}
\begin{equation}
    \rho \frac{d^2 h}{d\rho^2} + (2l + 2 - \rho) \frac{dh}{d\rho} + \left( \frac{m Z e^2}{4\pi\varepsilon_0 \hbar^2 K} - l - 1 \right) h = 0 %
\end{equation}
Questa è un'equazione molto simile a quella dell'oscillatore armonico. Cerchiamo una soluzione in serie di potenze: $h(\rho) = \sum_{i=0}^\infty a_i \rho^i$. %

\noindent
Come per l'oscillatore, una serie infinita produrrebbe $\infty$ TERMINI, portando a una funzione che diverge e non appartiene a $\mathcal{L}^2(\mathbb{R}^3)$ %. 
\textbf{Devo troncare la serie!} Questo è possibile se e solo se il coefficiente costante tra parentesi quadre si annulla per un certo intero positivo $n$: %
\begin{equation}
    \boxed{ \frac{m Z e^2}{4\pi\varepsilon_0 \hbar^2 K} = n } \quad \text{con } n = 1, 2, 3, \dots \text{ (\textbf{NUMERO QUANTICO PRINCIPALE})} %
\end{equation}

\subsection{Polinomi di Laguerre e Spettro Energetico} %

Il troncamento della serie genera i \textbf{Polinomi Associati di Laguerre}: %
\begin{equation*}
    h_{n-l-1}(\rho) = \sum_{i=0}^{n-l-1} \frac{(n+l)!}{(n-l-1-i)!(2l+1+i)! i!} (-\rho)^i %
\end{equation*}

\noindent
Perché il \textbf{grado del polinomio} ($n-l-1$) non sia negativo (mai $<0$), deve valere la condizione: %
\begin{equation}
    \boxed{ l < n } \quad \Rightarrow \quad \text{Dato } n, \text{ i valori permessi per } l \text{ sono } l = 0, 1, \dots, n-1 %
\end{equation}
Il valore $(n-l-1)$ rappresenta il \textbf{Numero di Zeri} (nodi) della parte radiale della funzione d'onda. %

Ricaviamo infine l'energia dalla condizione di troncamento: %
\begin{equation*}
    K = \frac{m Z e^2}{4\pi\varepsilon_0 \hbar^2 n} \quad \Rightarrow \quad K = \sqrt{-\frac{2mE}{\hbar^2}} \quad \Rightarrow \quad E_n = - \frac{\hbar^2 K^2}{2m} %
\end{equation*}
Sostituendo e semplificando, otteniamo lo spettro esatto dell'atomo di idrogeno: %
\begin{equation}
    \boxed{ E_n = - \frac{m Z^2 e^4}{2(4\pi\varepsilon_0)^2 \hbar^2} \frac{1}{n^2} = - Z^2 \frac{Ry}{n^2} } %
\end{equation}
Dove $Ry \simeq 13.6 \text{ eV}$ è l'Energia di Rydberg.
\textit{(Nota storica fondamentale: A differenza del modello di Bohr, qui NON ABBIAMO IPOTESI AD-HOC. L'energia quantizzata $1/n^2$ scaturisce puramente dalla richiesta matematica che la funzione d'onda sia a quadrato sommabile!)} 

\subsection{Livelli di Energia e Degenerazione} %

\textbf{Proprietà della Degenerazione:} %
\begin{enumerate}
    \item $E_n$ dipende \textbf{SOLO} da $n$. Questo accade unicamente perché il potenziale coulombiano ha la forma $V(r) \propto 1/r$. Negli atomi a più elettroni (dove c'è lo schermo elettronico e $V \neq V(r) \propto 1/r$), i livelli si sdoppiano e l'energia dipenderà sia da $n$ che da $l$: $E = E_{nl}$ (da cui l'ordine di riempimento degli orbitali nella tavola periodica!) %.
    \item Per ogni $l$, esistono $(2l+1)$ \textbf{STATI DEGENERI} (dovuti alle proiezioni $m$). Questo dipende dal fatto che il campo è centrale ($V=V(r)$ ha simmetria sferica). %
    \item \textbf{Rottura della Degenerazione:} Se voglio rimuovere la degenerazione sugli stati magnetici $m$, devo \textbf{rompere la simmetria sferica}. Ad esempio, applicando un campo elettrico o magnetico esterno uniassiale (Effetto Stark o Effetto Zeeman). %
\end{enumerate}

\section{Atomo di Idrogeno} %

Riprendiamo la soluzione asintotica dell'equazione radiale. Sostituendo la funzione ausiliaria $g(\rho)$ per trovare la parte radiale $R(\rho) = g(\rho)/r$, otteniamo: %
\begin{equation*}
    g(\rho) = \left( \frac{\rho}{2K} \right)^{l+1} e^{-\frac{\rho}{2}} h_{nl}(\rho) \quad \Rightarrow \quad R(\rho) \propto \left( \frac{\rho}{2K} \right)^l e^{-\frac{\rho}{2}} h_{nl}(\rho) %
\end{equation*}

\noindent
Ricordiamo il cambio di variabile $\rho = 2Kr$ e la condizione di quantizzazione imposta dal troncamento della serie di potenze: %
\begin{equation*}
    \frac{m Z e^2}{4\pi\varepsilon_0 K \hbar^2} = n %
\end{equation*}

Da questa uguaglianza possiamo ricavare esplicitamente il parametro $K$: %
\begin{equation*}
    K = \frac{m Z e^2}{4\pi\varepsilon_0 \hbar^2} \frac{1}{n} = \frac{1}{ \underbrace{ \frac{4\pi\varepsilon_0 \hbar^2}{m e^2} }_{\text{Raggio di Bohr}} } \frac{Z}{n} %
\end{equation*}

\noindent
Raggruppando le costanti fondamentali del sistema, emerge naturalmente l'espressione per il \textbf{Raggio di Bohr} ($a_0$). Sostituendolo all'interno, il parametro $K$ si semplifica notevolmente: %
\begin{equation}
    \boxed{ K = \frac{Z}{n a_0} } %
\end{equation}

\subsection{Le Autofunzioni dell'Idrogeno} %

Inseriamo ora questo parametro $K$ nell'espressione della parte radiale $R(r)$ per tornare alle coordinate fisiche spaziali $r$: %
\begin{equation*}
    R(r) \propto \left( \frac{2Zr}{n a_0} \right)^l e^{-\frac{Zr}{n a_0}} h_{nl}\left( \frac{2Zr}{n a_0} \right) %
\end{equation*}

\noindent
Aggiungendo la costante di normalizzazione $A_{nl}$ (che assorbe anche i fattori numerici ridondanti come il 2 e la $n$ all'interno delle parentesi dei polinomi), arriviamo all'espressione esatta finale per la \textbf{Parte Radiale}: %
\begin{equation}
    \boxed{ R_{nl}(r) = A_{nl} e^{-\frac{Zr}{n a_0}} \left( \frac{Zr}{a_0} \right)^l h_{nl}\left( \frac{Zr}{a_0} \right) } %
\end{equation}

\noindent
Unendo infine la parte radiale con la parte angolare (le Armoniche Sferiche $Y_l^m$), componiamo la funzione d'onda completa in 3D: %
\begin{equation}
    \boxed{ u_{nlm}(r, \theta, \varphi) = R_{nl}(r) Y_l^m(\theta, \varphi) } %
\end{equation}

\begin{quote}
    \textbf{Nota Fisica:} Queste sono le esatte Autofunzioni dell'atomo di Bohr, universalmente conosciute nella chimica e nella fisica quantistica moderna come \textbf{ORBITALI ATOMICI}! %
\end{quote}

\noindent
Tali autofunzioni costituiscono una base ortonormale in 3D: %
\begin{equation*}
    \int_{\mathbb{R}^3} u_{nlm}^* u_{n'l'm'} \, r^2 \sin\theta \, dr d\theta d\varphi = \delta_{nn'} \delta_{ll'} \delta_{mm'} %
\end{equation*}

\subsection{Esempi di Orbitali Idrogenoidi} %

\begin{itemize}
    \item \textbf{Stato 1s} ($n=1, l=0, m=0$): È una semplice funzione esponenziale decrescente (simmetria sferica perfetta). %
    \begin{equation*}
        u_{100} = \frac{1}{\sqrt{\pi}} \left(\frac{Z}{a_0}\right)^{3/2} e^{-Z r / a_0} %
    \end{equation*}
    
    \item \textbf{Stato 2s} ($n=2, l=0, m=0$): Compare un polinomio di grado $1$ ($n-l-1 = 2-0-1 = 1$), che genera un NODO radiale. %
    \begin{equation*}
        u_{200} = \frac{1}{2\sqrt{2\pi}} \left(\frac{Z}{a_0}\right)^{3/2} \underbrace{ \left(1 - \frac{Zr}{2a_0}\right) }_{\text{Causa il nodo}} e^{-Z r / 2a_0} %
    \end{equation*}

    \item \textbf{Stati 2p} ($n=2, l=1, m=0, \pm 1$): Non hanno nodi radiali ($2-1-1=0$), ma presentano la dipendenza angolare. %
    \begin{align*}
        u_{210} &= \frac{1}{4\sqrt{2\pi}} \left(\frac{Z}{a_0}\right)^{3/2} \left(\frac{Zr}{a_0}\right) e^{-Z r / 2a_0} \cos\theta \\ %
        u_{21\pm1} &= \frac{1}{8\sqrt{\pi}} \left(\frac{Z}{a_0}\right)^{3/2} \left(\frac{Zr}{a_0}\right) e^{-Z r / 2a_0} \sin\theta e^{\pm i\varphi} %
    \end{align*}
\end{itemize}

\noindent
\textit{Nomenclatura Chimica:} Gli stati con lo stesso numero quantico $n$ formano una \textbf{SHELL} (Guscio). Quelli con lo stesso $n$ e stesso $l$ formano una \textbf{SOTTOSHELL} (Sottoguscio). %

\subsection{Densità di Probabilità Radiale} %

La probabilità di trovare l'elettrone in un volume infinitesimo $d\tau$ è $|u_{nlm}|^2 d\tau$. Poiché $d\tau = r^2 dr \sin\theta d\theta d\varphi$, possiamo separare la probabilità in una componente radiale e una angolare: %
\begin{equation*}
    P_{d\tau} = \underbrace{ |R_{nl}(r)|^2 r^2 dr }_{\text{Prob. Radiale}} \times \underbrace{ |Y_l^m(\theta,\varphi)|^2 \sin\theta d\theta d\varphi }_{\text{Prob. Angolare } d\Omega} %
\end{equation*}

\begin{quote}
    \textbf{NON ESISTE PIÙ L'ORBITA:} L'elettrone non è un pianeta che gira attorno al Sole. È una \textit{nuvola di probabilità} statica attorno al nucleo! %
\end{quote}

\noindent
Attenzione: La densità di probabilità radiale $P_{nl}(r)$ (la probabilità di trovare l'elettrone a una distanza compresa tra $r$ e $r+dr$ indipendentemente dall'angolo) \textbf{NON È} semplicemente il modulo quadro di $R$, ma include l'elemento di volume sferico $r^2$: %
\begin{equation}
    \boxed{ P_{nl}(r) = |R_{nl}(r)|^2 r^2 } %
\end{equation}

\begin{enumerate}
    \item \textbf{Nodi Radiali:} Il numero di punti in cui la probabilità si annulla (escludendo l'origine e l'infinito) è esattamente: %
    \begin{equation*}
        \text{Nodi} = n - l - 1 %
    \end{equation*}
    \textit{(Es: il 1s ha 0 nodi, il 2s ha 1 nodo, il 3s ha 2 nodi, il 2p ha 0 nodi).} %

    \item \textbf{Valore Medio della Distanza ($\langle r \rangle$):} Il valore atteso della distanza elettrone-nucleo è: %
    \begin{equation*}
        \langle r \rangle = \int_0^{+\infty} r P_{nl}(r) \, dr = \frac{n^2 a_0}{Z} \left\{ 1 + \frac{1}{2} \left[ 1 - \frac{l(l+1)}{n^2} \right] \right\} %
    \end{equation*}
    Per $l \approx n-1$, si vede che $\langle r \rangle \approx \frac{n^2 a_0}{Z}$, che è esattamente il raggio delle orbite calcolato semiclassicamente da Bohr! Il limite classico emerge naturalmente. %

    \begin{figure}[H]
        \centering
        \includegraphics[width=0.3\linewidth]{immagini/2026-04-24-17-18-37.png}
        \end{figure}

    \item \textbf{Comportamento nell'Origine ($r \to 0$):} %
    Cosa succede vicino al nucleo? Sappiamo che il potenziale efficace è $V_{eff} = \frac{\hbar^2 l(l+1)}{2mr^2} - \frac{Z e^2}{4\pi\varepsilon_0 r}$. %
    \begin{itemize}
        \item Se $l \ge 1$ (orbitali p, d, f): La barriera centrifuga $1/r^2$ spinge via l'elettrone. La probabilità nell'origine è \textbf{NULLA}. %
        \item Se $l = 0$ (orbitali s): La barriera centrifuga si annulla! Rimane solo l'attrazione coulombiana pura. %
    \end{itemize}
    \textbf{Conclusione chimica vitale:} SOLO gli orbitali di tipo \textit{s} hanno una probabilità non nulla di penetrare e trovarsi nel nucleo dell'atomo (fenomeno responsabile della cattura elettronica e di effetti spettroscopici fini). %
\end{enumerate}
