\section{Formalismo Lagrangiano}

Dalla Legge di Newton:
\begin{equation}
    F_x^{tot} = m a_x \Rightarrow m \frac{d^2x(t)}{dt^2} = F_x^{tot} \quad \text{con condizioni iniziali} \begin{cases} x(0) \\ \frac{dx}{dt}(0) \end{cases} \Rightarrow x = x(t) \text{ (Legge del moto)}
\end{equation}


\begin{figure}[h]
	\centering
	\includegraphics[width=0.3\textwidth]{immagini/2026-03-01-20-21-30.png}
\end{figure}

\noindent
\textbf{GdL}: n° di parametri indipendenti per descrivere interamente il sistema. \\
$n \text{ GdL} \Rightarrow n$ Coordinate Generalizzate $q_1, q_2, q_3, \dots, q_n \Rightarrow \dot{q}_i = \frac{dq_i}{dt}$ (velocità generalizzate). \\
\textbf{Scopo}: trovare $q = q(t)$.

\subsection{Lagrangiana}
\begin{equation}
    \mathcal{L} = \mathcal{L}(q_1, \dots, q_n ; \dot{q}_1, \dots, \dot{q}_n, t) = \underbrace{K(q_1\dots q_n, \dot{q}_1\dots \dot{q}_n)}_{\text{Energ. Cinet.}} - \underbrace{V(q_1\dots q_n, t)}_{\text{Energ. pot.}}
\end{equation}
$\mathcal{L} = K(q, \dot{q}) - V(q, t) \longrightarrow$ Sistemi conservativi.

\noindent
$\rightarrow$ Sistema di $N$ punti materiali: $K = \sum_{i=1}^N \frac{1}{2} m_i |\vec{v}_i|^2$ (dove $\vec{v}_i \rightarrow \dot{q}_{ix}, \dot{q}_{iy}, \dot{q}_{iz}$) e $V = V(\vec{r}_1 \dots \vec{r}_n, t)$.

\subsection{Azione e Principio di Hamilton}
L'\textbf{Azione} è definita come:
\begin{equation}
    S = \int_{t_1}^{t_2} \mathcal{L}(q(t), \dot{q}(t), t) dt \quad \text{dove } [S] = E \cdot t \Rightarrow \text{Momento angolare}
\end{equation}

\begin{figure}[h]
	\centering
	\includegraphics[width=0.3\textwidth]{immagini/2026-03-01-20-35-48.png}
\end{figure}

\noindent
$S = \int_{t_1}^{t_2} \mathcal{L}(q, \dot{q}, t) dt \rightarrow \underline{\text{FUNZIONALE}}$ (la $q$ non è univocamente definita, si valuta un'intera famiglia di possibili $q(t)$).

\vspace{3cm}

\noindent
\textbf{PRINCIPIO di HAMILTON (Azione stazionaria):}
\begin{equation}
    \boxed{\delta S = \delta \int_{t_1}^{t_2} \mathcal{L}(q, \dot{q}, t) dt = 0} \quad \text{la } q \text{ vera è tale per cui } S \text{ è stazionaria.}
\end{equation}
\qquad \qquad \qquad \quad $\hookrightarrow$ differenziale di un funzionale.

\textbf{Es. Derivazione:} consideriamo $\mathcal{L} = K(q, \dot{q}) - V(q)$.
\begin{align*}
    \delta S &= \int_{t_1}^{t_2} \left( \frac{\partial \mathcal{L}}{\partial q}\delta q + \underbrace{\frac{\partial \mathcal{L}}{\partial \dot{q}}\delta \dot{q}}_{\text{diff. tra due } q(t) \text{ inf. vicine}} \right) dt = \int_{t_1}^{t_2} \left( \frac{\partial \mathcal{L}}{\partial q}\delta q + \frac{\partial \mathcal{L}}{\partial \dot{q}} \frac{d}{dt}(\delta q) \right) dt = 0
\end{align*}

Integriamo per parti il secondo termine:
\begin{equation*}
    \int_{t_1}^{t_2} \underbrace{\frac{\partial \mathcal{L}}{\partial \dot{q}}}_{f(t)} \underbrace{\frac{d}{dt}(\delta q)}_{g'(t)} dt = \cancelto{0}{\left[ \frac{\partial \mathcal{L}}{\partial \dot{q}} \delta q \right]_{t_1}^{t_2}} - \int_{t_1}^{t_2} \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial \dot{q}}\right) \delta q \, dt
\end{equation*}
Il termine di bordo si annulla perché le variazioni agli estremi sono nulle: $\delta q(t_1) = \delta q(t_2) = 0$.

Sostituendo indietro:
\begin{equation}
    \delta S = \int_{t_1}^{t_2} \underbrace{\left[ \frac{\partial \mathcal{L}}{\partial q} - \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial \dot{q}}\right) \right]}_{= 0} \overbrace{\delta q}^{\text{arbitrario}} dt = 0 \quad \Rightarrow \quad \boxed{ \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial \dot{q}_i}\right) = \frac{\partial \mathcal{L}}{\partial q_i} }
\end{equation}
Queste sono le \textbf{Eq. di Lagrange}.

\subsection{Momento Canonico e Coordinate Cicliche}
Definiamo il momento canonico coniugato alla coordinata $q_k$:
\begin{equation}
    p_k = \frac{\partial \mathcal{L}}{\partial \dot{q}_k} \quad \Rightarrow \quad \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial \dot{q}_k}\right) = \frac{\partial \mathcal{L}}{\partial q_k} \quad \Rightarrow \quad \frac{d}{dt}(p_k) = \frac{\partial \mathcal{L}}{\partial q_k}
\end{equation}
Questo è l'analogo della Legge di Newton. Se la Lagrangiana non dipende esplicitamente da una coordinata $q_k$ ($\mathcal{L} \neq \mathcal{L}(q_k)$):
\begin{equation}
    \frac{\partial \mathcal{L}}{\partial q_k} = 0 \quad \Rightarrow \quad \frac{d}{dt}(p_k) = 0 \quad \Rightarrow \quad p_k = \text{cost} \quad \Rightarrow \quad \ulcorner q_k \text{ è CICLICA} \lrcorner
\end{equation}

\section{Formalismo Hamiltoniano}

Partiamo dalle $n$ Equazioni di Lagrange del II ordine:
\begin{equation*}
    \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial \dot{q}_i}\right) = \frac{\partial \mathcal{L}}{\partial q_i} \quad i=1\dots n \qquad \text{con cond. iniziali } \begin{cases} q_i(0) \\ \dot{q}_i(0) = \frac{dq_i}{dt}(0) \end{cases}
\end{equation*}

\noindent
Passiamo da $n$ variabili indipendenti a $2n$ variabili indipendenti tramite la trasformazione di Legendre: $(q_i, \dot{q}_i) \rightleftarrows (q_i, p_i)$. Definiamo l'Hamiltoniana:
\begin{equation}
    H(q,p) = \sum_{i=1}^n p_i \dot{q}_i - \mathcal{L}(q, \dot{q})
\end{equation}
$\rightarrow$ Se il sistema è conservativo $\Rightarrow \boxed{H(q,p) = K(q,p) + V(q)}$

\noindent
Le equazioni del moto diventano le \textbf{Equazioni di Hamilton} (ho raddoppiato le variabili, ma ora le equazioni sono del I ordine):
\begin{equation}
    \text{Cond. iniziali: } \begin{cases} q_i(0) \\ p_i(0) \end{cases} \Rightarrow \quad \begin{cases} \dot{q}_i = \frac{\partial H}{\partial p_i} \\ \dot{p}_i = -\frac{\partial H}{\partial q_i} \end{cases} \quad i=1\dots n
\end{equation}

\subsubsection{Es.: $H(q,p)$ per particella con $V(x,y,z)$}
\begin{equation*}
    \mathcal{L} = \frac{1}{2}m(\dot{x}^2 + \dot{y}^2 + \dot{z}^2) - V(x,y,z) \quad \Rightarrow \quad H = \sum_{i=x,y,z} p_i \dot{q}_i - \mathcal{L}
\end{equation*}
Ricordando che $p_i = \frac{\partial \mathcal{L}}{\partial \dot{q}_i} \Rightarrow p_x = m\dot{x} \Rightarrow \dot{x} = \frac{p_x}{m}$:
\begin{align*}
    H &= p_x \dot{x} + p_y \dot{y} + p_z \dot{z} - \frac{1}{2}m(\dot{x}^2 + \dot{y}^2 + \dot{z}^2) + V(x,y,z) = \\
      &= \frac{p_x^2}{m} + \frac{p_y^2}{m} + \frac{p_z^2}{m} - \frac{1}{2}m\left(\frac{p_x^2}{m^2} + \frac{p_y^2}{m^2} + \frac{p_z^2}{m^2}\right) + V(x,y,z) = \\
      &= \underbrace{\frac{1}{2m}(p_x^2 + p_y^2 + p_z^2)}_{K(p)} + \underbrace{V(x,y,z)}_{V(q)}
\end{align*}

\subsection{Es.: $\mathcal{L}$ di una particella immersa in un campo Elettromagnetico}

\textbf{Equazioni di Maxwell e Potenziali:}
\begin{align*}
    \text{div} \vec{E} &= \frac{\rho}{\varepsilon_0} & \text{div} \vec{B} &= 0 \quad \Rightarrow \quad \vec{B} = \text{rot} \vec{A} \\
    \text{rot} \vec{E} &= -\frac{\partial \vec{B}}{\partial t} \quad \Rightarrow \quad \vec{E} = -\nabla \varphi - \frac{\partial \vec{A}}{\partial t} & \text{rot} \vec{B} &= \mu_0 \vec{J} + \mu_0\varepsilon_0 \frac{\partial \vec{E}}{\partial t}
\end{align*}
Vettore potenziale e potenziale scalare: $\vec{A} = \begin{bmatrix} A_x(x,y,z,t) \\ A_y(x,y,z,t) \\ A_z(x,y,z,t) \end{bmatrix}$ e $\varphi = \varphi(x,y,z,t)$. \\
Invarianza di \textbf{Gauge}: $\vec{A} \rightarrow \vec{A}' = \vec{A} + \nabla f$ e $\varphi \rightarrow \varphi' = \varphi - \frac{\partial f}{\partial t}$.\\

\noindent
\textbf{La Lagrangiana} è data da:
\begin{equation}
    \boxed{ \mathcal{L} = \underbrace{\frac{1}{2}m|\vec{v}|^2}_{E_K} - \underbrace{e\varphi + e\vec{v}\cdot\vec{A}}_{\vec{F}_{el} \text{ e } \vec{J}\cdot\vec{A}} }
\end{equation}
Espandendo in componenti:
\begin{equation*}
    \mathcal{L} = \frac{1}{2}m(v_x^2 + v_y^2 + v_z^2) - e\varphi + ev_x A_x + ev_y A_y + ev_z A_z
\end{equation*}

Applicando l'equazione di Lagrange per la coordinata $x$ ($q_i = x, \dot{q}_i = \dot{x} = v_x$):
\begin{equation}
    \boxed{ \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial v_x}\right)^{(1)} = \left(\frac{\partial \mathcal{L}}{\partial x}\right)^{(2)} }
\end{equation}

\textbf{Derivata (1):}
\begin{align*}
    \frac{d}{dt}\left(\frac{\partial \mathcal{L}}{\partial v_x}\right) &= \frac{d}{dt}(mv_x + eA_x) = ma_x + e\frac{dA_x}{dt} = \\
    &= \ulcorner ma_x + e\left( \frac{\partial A_x}{\partial x}v_x + \frac{\partial A_x}{\partial y}v_y + \frac{\partial A_x}{\partial z}v_z + \frac{\partial A_x}{\partial t} \right) \lrcorner
\end{align*}

\textbf{Derivata (2):}
\begin{equation*}
    \frac{\partial \mathcal{L}}{\partial x} = \ulcorner -e\frac{\partial \varphi}{\partial x} + e\left( \frac{\partial A_x}{\partial x}v_x + \frac{\partial A_y}{\partial x}v_y + \frac{\partial A_z}{\partial x}v_z \right) \lrcorner
\end{equation*}

Uguagliando (1) e (2) ed elidendo i termini $e\frac{\partial A_x}{\partial x}v_x$:
\begin{align*}
    ma_x &= -e\left( \cancel{\frac{\partial A_x}{\partial x}v_x} + \frac{\partial A_x}{\partial y}v_y + \frac{\partial A_x}{\partial z}v_z + \frac{\partial A_x}{\partial t} \right) - e\frac{\partial \varphi}{\partial x} + e\left( \cancel{\frac{\partial A_x}{\partial x}v_x} + \frac{\partial A_y}{\partial x}v_y + \frac{\partial A_z}{\partial x}v_z \right) \\
    ma_x &= -e\underbrace{\left[ \frac{\partial \varphi}{\partial x} + \frac{\partial A_x}{\partial t} \right]}_{-eE_x} + e\left[ v_y \underbrace{\left( \frac{\partial A_y}{\partial x} - \frac{\partial A_x}{\partial y} \right)}_{(\text{rot}\vec{A})_z = B_z} + v_z \underbrace{\left( \frac{\partial A_z}{\partial x} - \frac{\partial A_x}{\partial z} \right)}_{-(\text{rot}\vec{A})_y = -B_y} \right] \\
    ma_x &= -eE_x + e(v_y B_z - v_z B_y) = -eE_x + e(\vec{v} \wedge \vec{B})_x
\end{align*}
Generalizzando in 3D, ritroviamo la \textbf{Forza di Lorentz}:
\begin{equation*}
    \ulcorner m\vec{a} = -e\vec{E} + e(\vec{v} \wedge \vec{B}) \lrcorner \quad \text{(Newton con Forza di Lorentz)}
\end{equation*}\\

\noindent
\textbf{Hamiltoniana del campo E.M.:}
Troviamo il momento canonico $p_x = \frac{\partial \mathcal{L}}{\partial \dot{x}} = m\dot{x} + eA_x \Rightarrow \ulcorner \vec{p} = m\vec{v} + e\vec{A} \lrcorner \Rightarrow m\vec{v} = \vec{p} - e\vec{A}$.
Poiché $H = K + V$:
\begin{equation}
    \boxed{ H = \underbrace{\frac{(\vec{p} - e\vec{A})^2}{2m}}_{\text{solo dipendente da } \vec{v}} + \underbrace{e\varphi}_{\text{en. pot. elettrost.}} }
\end{equation}

\section{Fondamenti di Meccanica Statistica: Ensemble}

Prendiamo come esempio un \textbf{Gas Perfetto Monoatomico}:
Consideriamo 1 mole di gas, che contiene un numero di Avogadro di particelle ($N_A \simeq 10^{24}$). 
Macroscopicamente il sistema è descritto da $p, V, T$.
Microscopicamente, per ogni particella abbiamo coordinate $(x,y,z)$ e momenti $(p_x, p_y, p_z)$, risultando in $\simeq 10^{24}$ Equazioni di Hamilton:
\begin{equation*}
    \begin{cases}
        \dot{q}_i = \frac{\partial H}{\partial p_i} \\
        \dot{p}_i = -\frac{\partial H}{\partial q_i}
    \end{cases} \quad \text{con cond. iniziali } (q_i(0), p_i(0))
\end{equation*}

\noindent
Data una generica grandezza macroscopica $G = G(x_1 \dots x_N, p_1 \dots p_N) = G(q,p)$, vorremmo calcolarne la \textbf{media temporale}:
\begin{equation}
    \bar{G} = \frac{1}{T} \int_0^T G(q(t), p(t)) \, dt
\end{equation}
Tuttavia, questo calcolo è \textbf{NON FATTIBILE} perché richiederebbe di conoscere $\sim 10^{24}$ condizioni iniziali $\begin{cases} q_i(0) \\ p_i(0) \end{cases}$.

\subsection{Il Concetto di Ensemble}

Supponiamo di costruire $M$ repliche del barattolo originale. Tutte le repliche hanno le \textbf{stesse condizioni macroscopiche} ($p, V, T$) ma \textbf{condizioni iniziali (microscopiche) diverse}.
Facendo tendere il numero di repliche all'infinito ($M \to \infty$), otteniamo un \textbf{ENSEMBLE}.

\begin{figure}[h]
	\centering
	\includegraphics[width=0.25\textwidth]{immagini/2026-03-04-15-00-35.png}
\end{figure}

\noindent
Possiamo ora definire la \textbf{Media Pesata sull'Ensemble}:
\begin{equation}
    \langle G \rangle = \frac{1}{M} \sum_{i=1}^M G_i
\end{equation}

\noindent
Nello spazio delle fasi, queste innumerevoli repliche creano una densità continua. Definiamo la densità di sistemi in $\Gamma$:
\begin{equation*}
    \rho(q,p) = \frac{\text{\# sistemi che hanno } (q_i, p_i)}{\text{Vol. in } \Gamma} \quad \Rightarrow \quad dM(q_i, p_i) = \rho(q_i, p_i) d\Gamma
\end{equation*}

\begin{figure}[h]
	\centering
	\includegraphics[width=0.25\textwidth]{immagini/2026-03-04-14-59-22.png}
\end{figure}

Passando al continuo, la media della grandezza $G$ diventa:
\begin{equation}
    \langle G \rangle = \frac{\int G(q, p) \rho(q, p) \, dq \, dp}{\int \rho(q, p) \, dq \, dp}
\end{equation}

\noindent
Se normalizziamo il numero totale di sistemi a 1, ovvero $\int \rho(q, p) \, dq \, dp = 1$, la densità $\rho(q,p)$ diventa una vera e propria \textbf{Densità di Probabilità Classica}. L'equazione si semplifica in:
\begin{equation}
    \boxed{ \langle G \rangle = \int G(q, p) \rho(q, p) \, d\Gamma }
\end{equation}

\subsection{Ipotesi Ergodica}
L'assunzione fondamentale che permette di collegare la media temporale con quella statistica è l'\textbf{Ipotesi Ergodica}:
\begin{equation}
    \boxed{ \bar{G} = \langle G \rangle }
\end{equation}
(La media temporale di un singolo sistema per $T \to \infty$ è uguale alla media sull'ensemble a un dato istante $t$).

\noindent
Il punto fondamentale della Meccanica Statistica diventa quindi trovare la distribuzione di probabilità corretta: \textbf{$\rho(q,p) = ?$}.

\subsection{Teorema di Liouville}

Il punto fondamentale della meccanica statistica è trovare la densità $\rho(q,p)$. Il \textbf{Teorema di Liouville} stabilisce che per un sistema in equilibrio la densità di probabilità nello spazio delle fasi è una costante del moto:
\begin{equation}
    \boxed{ \frac{d}{dt}[\rho(q,p)] = 0 } \quad \text{STAZIONARIA}
\end{equation}
Di conseguenza, la densità di probabilità può dipendere dalle coordinate e dai momenti solo tramite l'energia del sistema:
\begin{figure}[ht]

    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        \begin{equation*}
            \rho(q,p) \Rightarrow \rho(E) = \rho[E(q,p)]
        \end{equation*}
    \end{minipage}
\end{figure}

\noindent
Verifichiamo l'Ipotesi Ergodica ($\bar{v} = \langle v \rangle$) confrontando i due approcci per una grandezza come la velocità:
\begin{enumerate}
    \item \textbf{Media temporale:} $\bar{v} = \frac{1}{T} \int_0^T v(q_1, q_2, p_1, p_2) \, dt$
    \item \textbf{M Repliche (Ensemble):} $\langle v \rangle = \frac{\int v(q,p) \rho(q,p) \, dq \, dp}{\int \rho(q,p) \, dq \, dp}$
\end{enumerate}

\section{Distribuzione di Boltzmann e Funzione di Partizione}

Riprendiamo l'\textbf{Ensemble Canonico} (Sistemi in equilibrio termico). La probabilità di trovare il sistema con energia $E$ è data dalla \textbf{Distribuzione di Boltzmann}:
\begin{equation*}
    \rho(E) = c e^{-\beta E} = c e^{-\frac{E}{k_B T}}
\end{equation*}
dove $c = \text{cost}$, $\beta = \frac{1}{k_B T}$, e $k_B = 1,38 \cdot 10^{-23} \text{ J/K}$ (Costante di Boltzmann).\\

\noindent
\textit{Esempio dimensionale:} A temperatura ambiente $T = 300 \text{ K}$:
\begin{equation*}
    k_B T = 4,14 \cdot 10^{-21} \text{ J}
\end{equation*}
Introducendo l'elettronvolt ($1 \text{ eV} = 1,6 \cdot 10^{-19} \text{ J}$, l'energia cinetica acquisita da un elettrone sotto una differenza di potenziale di 1 Volt):
\begin{equation*}
    k_B T \simeq 2,5 \cdot 10^{-2} \text{ eV} = 25 \text{ meV}
\end{equation*}

Affinché la densità di probabilità sia normalizzata a 1 nello spazio delle fasi $\Gamma$:
\begin{equation*}
    1 = \int \rho(q,p) \, dq \, dp = \int c e^{-\beta E} \, dq \, dp \quad \Rightarrow \quad c = \frac{1}{\int e^{-\beta E} \, dq \, dp}
\end{equation*}

Definiamo il denominatore dell'espressione come \textbf{Funzione di Partizione} ($Z$):
\begin{equation}
    \boxed{ Z = \int e^{-\beta E} \, dq \, dp }
\end{equation}
(Se $c$ ha questo valore, allora $\rho$ è una vera e propria densità di probabilità).\\

\noindent
La probabilità normalizzata diventa quindi: $\rho(E) = \frac{e^{-\beta E}}{Z}$. 
Di conseguenza, il valore medio di una qualsiasi grandezza $G$ è:
\begin{equation}
    \boxed{ \langle G \rangle = \frac{\int G(q,p) e^{-\beta E} \, dq \, dp}{Z} }
\end{equation}

\subsection{Energia Media e Sottosistemi}

Se consideriamo due sottosistemi $S_1$ e $S_2$ distinti e indipendenti, l'energia si somma: $E_{1+2} = E_1 + E_2$.
La funzione di partizione totale si fattorizza nel prodotto delle singole funzioni di partizione:
\begin{figure}[ht]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-03-04-15-59-39.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        \begin{align*}
            Z_{1+2} &= \int e^{-\beta E_{1+2}} \, dq_1 \, dp_1 \, dq_2 \, dp_2 = \int e^{-\beta(E_1 + E_2)} \, dq_1 \, dp_1 \, dq_2 \, dp_2 = \\
                    &= \left( \int e^{-\beta E_1} \, dq_1 \, dp_1 \right) \cdot \left( \int e^{-\beta E_2} \, dq_2 \, dp_2 \right) = Z_1 \cdot Z_2
        \end{align*}
    \end{minipage}
\end{figure}

Generalizzando a $N$ sottosistemi indipendenti: $Z_N = \prod_{j=1}^N Z_j$.\\

\noindent
Calcoliamo ora l'Energia Media $\langle E \rangle = U$ sfruttando $Z$:
\begin{equation*}
    \frac{\partial}{\partial \beta} \ln(Z) = \frac{1}{Z} \frac{\partial Z}{\partial \beta} = \frac{1}{Z} \frac{\partial}{\partial \beta} \int e^{-\beta E} \, dq \, dp = \frac{1}{Z} \int \frac{\partial}{\partial \beta}(e^{-\beta E}) \, dq \, dp = -\frac{1}{Z} \int E e^{-\beta E} \, dq \, dp
\end{equation*}
Riconosciamo nell'ultimo termine la definizione di valore medio, ottenendo la relazione fondamentale:
\begin{equation}
    \boxed{ \langle E \rangle = U = -\frac{\partial}{\partial \beta} \ln Z }
\end{equation}

\section{Teorema di Equipartizione dell'Energia}
(Valido per sistemi all'equilibrio termico).

\noindent
Supponiamo che l'energia dipenda in modo quadratico da una delle variabili dello spazio delle fasi (chiamiamola $x$, che può essere una coordinata $q_i$ o un momento $p_i$) e che questa sia indipendente dal resto delle variabili:
\begin{equation*}
    E = a x^2 + f(q,p)_{q \neq x}
\end{equation*}
Il valore medio dell'energia sarà $\langle E \rangle = \langle a x^2 \rangle + \langle f(q,p) \rangle$.

Calcoliamo esplicitamente $\langle a x^2 \rangle$:
\begin{align*}
    \langle a x^2 \rangle &= \frac{\int a x^2 e^{-\beta E} \, dq \, dp}{Z} = \frac{\int a x^2 e^{-\beta(a x^2 + f)} \, dq \, dp}{\int e^{-\beta(a x^2 + f)} \, dq \, dp} = \\
    &= \frac{\int a x^2 e^{-\beta a x^2} \, dx \cdot \cancel{\int e^{-\beta f} \, d(q,p)_{q \neq x}}}{\int e^{-\beta a x^2} \, dx \cdot \cancel{\int e^{-\beta f} \, d(q,p)_{q \neq x}}}
\end{align*}

Questo è l'integrale di una gaussiana. Possiamo usare il "trucco" della derivata del logaritmo:
\begin{align*}
    \frac{\int a x^2 e^{-\beta a x^2} \, dx}{\int e^{-\beta a x^2} \, dx} &= -\frac{\partial}{\partial \beta} \ln \left[ \int e^{-\beta a x^2} \, dx \right] = -\frac{\partial}{\partial \beta} \ln \left[ \sqrt{\frac{\pi}{\beta a}} \right] = \\
    &= -\frac{1}{2} \frac{\partial}{\partial \beta} [\ln \pi - \ln \beta - \ln a] = -\frac{1}{2} \left(-\frac{1}{\beta}\right) = \frac{1}{2\beta} = \frac{1}{2} k_B T
\end{align*}
Ne consegue che ogni termine quadratico indipendente "assorbe" un'energia media pari a:
\begin{equation}
    \boxed{ \langle a x^2 \rangle = \frac{1}{2} k_B T }
\end{equation}
\textit{Esempio:} Se $E = a x^2 + b p_y^2 + c x_1 + d p_1 \Rightarrow \langle E \rangle = \frac{1}{2} k_B T + \frac{1}{2} k_B T + \langle c x_1 + d p_1 \rangle$.

\section{Applicazione: Gas Perfetto Monoatomico}

Consideriamo la singola molecola di massa $m$ (ha solo energia cinetica traslazionale):

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.6\textwidth]{immagini/2026-03-04-16-28-01.png}
\end{figure}

\begin{equation*}
    E_{1mol} = \frac{1}{2m}(p_x^2 + p_y^2 + p_z^2)
\end{equation*}
La funzione di partizione di questa singola particella ($Z_{mol}$) è:
\begin{align*}
    Z_{mol} &= \int e^{-\beta E} \, dq \, dp = \int e^{-\beta \left[\frac{1}{2m}(p_x^2 + p_y^2 + p_z^2)\right]} \, dx \, dy \, dz \, dp_x \, dp_y \, dp_z = \\
            &= V \cdot \left( \int e^{-\beta \frac{p^2}{2m}} \, dp \right)^3 = V \cdot (2\pi m k_B T)^{3/2}
\end{align*}

\noindent
Se consideriamo una mole di gas perfetto, ovvero $N_A$ molecole come $N_A$ sottosistemi \textbf{NON INTERAGENTI}, la funzione di partizione del gas si fattorizza:
\begin{equation}
    \boxed{ Z_{gas} = \prod_{i=1}^{N_A} Z_{moli} = \left[ V (2\pi m k_B T)^{3/2} \right]^{N_A} = V^{N_A} (2\pi m k_B T)^{\frac{3N_A}{2}} }
\end{equation}

Troviamo l'Energia Interna del gas ($U_{gas}$) derivando $Z_{gas}$:
\begin{align*}
    \langle E \rangle = U_{gas} &= -\frac{\partial}{\partial \beta} \ln(Z_{gas}) = -\frac{\partial}{\partial \beta} \ln \left[ V^{N_A} \left( \frac{2\pi m}{\beta} \right)^{\frac{3N_A}{2}} \right] = -\frac{\partial}{\partial \beta} \ln \left[ \frac{V^{2/3} 2\pi m}{\beta} \right]^{\frac{3N_A}{2}} = \\
    &= -\frac{3N_A}{2} \frac{\partial}{\partial \beta} \ln \left( \frac{V^{2/3} 2\pi m}{\beta} \right) = -\frac{3N_A}{2} \left( \frac{1}{\frac{V^{2/3} 2\pi m}{\beta}} \right) \cdot \left( -\frac{V^{2/3} 2\pi m}{\beta^2} \right) = \\
    &= \frac{3N_A}{2} \frac{1}{\beta} = \frac{3}{2} N_A k_B T = \frac{3}{2} R T
\end{align*}

*Possiamo ottenere lo stesso risultato istantaneamente dal Teorema di Equipartizione:*
\begin{equation*}
    E_{1mol} = \frac{p_x^2}{2m} + \frac{p_y^2}{2m} + \frac{p_z^2}{2m} \quad \Rightarrow \quad \langle E_{1mol} \rangle = \underbrace{\frac{1}{2} k_B T}_x + \underbrace{\frac{1}{2} k_B T}_y + \underbrace{\frac{1}{2} k_B T}_z = \frac{3}{2} k_B T
\end{equation*}
Da cui, per l'intero gas con $N_A$ particelle:
\begin{equation}
    \langle E_{gas} \rangle = \langle E_{1mol} \rangle N_A = \frac{3}{2} N_A k_B T = \frac{3}{2} R T
\end{equation}

Infine, il calore specifico a volume costante si calcola come:
\begin{equation}
    C_v = \left( \frac{\partial U}{\partial T} \right)_V = \frac{3}{2} R
\end{equation}

\section{Sezione d'Urto ($\sigma$)} % [cite: 583]

Per misurare le proprietà della materia, si utilizzano \textbf{SONDE} (es. fascio di elettroni o raggi X) indirizzate contro un sottile foglio di materiale che funge da bersaglio %[cite: 584, 585].

\begin{figure}[H]
    \centering
    \includegraphics[width=0.7\linewidth]{immagini/2026-03-04-17-00-31.png}
    \end{figure}

\noindent
Sia $n_0$ il numero di atomi del bersaglio (che agiscono da centri diffusori) e $N_0$ l'intensità del fascio primario incidente %[cite: 586, 589, 590, 591].
Oltrepassato il bersaglio, il numero di particelle trasmesse è $N < N_0$, mentre il numero di particelle "rimosse" dal fascio è $R = N_0 - N$ %[cite: 593, 594].

\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-03-04-17-01-29.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        Definiamo la \textbf{sezione d'urto} $\sigma$ (misurata in $\text{m}^2$) %[cite: 597].
        La probabilità di un evento di rimozione dal fascio è il rapporto tra l'area coperta dai centri diffusori e l'area totale $A$ del bersaglio:
        \begin{equation}
            P = \frac{n_0 \cdot \sigma}{A} % [cite: 600, 601]
        \end{equation}
    \end{minipage}
\end{figure}
\textit{Nota: se $n_0 \sigma = A \Rightarrow$ non passa nulla.} % [cite: 598, 603]

Il numero di particelle rimosse sarà quindi:
\begin{equation*}
    R = P \cdot N_0 = \frac{n_0 \sigma}{A} N_0 % [cite: 605]
\end{equation*}

La sezione d'urto totale è la somma delle sezioni d'urto dei singoli processi:
\begin{equation*}
    \sigma_{\text{TOT}} = \sigma_{\text{FOTOEL}} + \sigma_{\text{COMP}} + \sigma_{\text{RAY}} \quad \Rightarrow \quad \sigma_{\text{TOT}} = \sum \sigma_i % [cite: 606]
\end{equation*}

        \begin{equation}
            -dN = N(x+dx) - N(x) = \sigma \rho N(x) dx \quad \Rightarrow \quad \frac{dN}{N(x)} = -\sigma \rho \, dx % [cite: 615, 616]

\begin{equation}

\end{equation}
dove $\Lambda = \frac{1}{\sigma \rho}$ è il \textbf{libero cammino medio} %[cite: 628, 629].

        \begin{itemize}
            \item $b = \text{parametro d'impatto}$ %[cite: 634, 635].
            \item $\theta = \text{angolo di diffusione}$ %[cite: 635].
        \end{itemize}
    \end{minipage}
\end{figure}

L'angolo di diffusione dipende dal parametro d'impatto: $\theta = \theta(b)$

\noindent
Dalla geometria della collisione (con raggio del centro diffusore $R$), abbiamo $b = R \sin\alpha$ %[cite: 638]. 
Osservando gli angoli, notiamo che:
\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering

    \begin{minipage}{0.65\textwidth}
        \begin{equation*}
            2\alpha + \theta = \pi \quad \Rightarrow \quad \alpha = \frac{\pi}{2} - \frac{\theta}{2} % [cite: 637, 638]
        \end{equation*}
        Sostituendo $\alpha$:
        \begin{equation}
            b = R \sin\left(\frac{\pi}{2} - \frac{\theta}{2}\right) = R \cos\left(\frac{\theta}{2}\right) % [cite: 638]
        \end{equation}
    \end{minipage}
\end{figure}

Invertendo la relazione, otteniamo l'angolo di scattering in funzione del parametro d'impatto:
\begin{equation}
    \boxed{
    \theta = \begin{cases} 
    2 \arccos\left(\frac{b}{R}\right) & \text{se } b \le R \\ 
    0 & \text{altrimenti (non c'è urto)} 
    \end{cases}
    }
\end{equation}

\subsection{Sezione d'Urto Differenziale}

\begin{figure}[H]
    \centering
    \begin{minipage}{0.5\textwidth}
        \centering
        \includegraphics[width=0.9\linewidth]{immagini/2026-03-05-17-34-27.png}
    \end{minipage}\hfill
    \begin{minipage}{0.5\textwidth}
        \includegraphics[width=0.8\linewidth]{immagini/2026-03-05-17-34-54.png}
    \end{minipage}
\end{figure}

Considerando la simmetria cilindrica, l'area dell'anello infinitesimo associato al parametro d'impatto è:
\begin{equation}
    \boxed{ d\sigma = 2\pi b \, db } \quad \leftarrow \text{SEZIONE D'URTO INFINITESIMA}
\end{equation}

\noindent
Sulla sfera di raggio $r$, l'elemento di superficie $dS$ spazzato dall'angolo solido $d\Omega$ è:
\begin{equation*}
    dS = r \cdot r \, d\theta = 2\pi r^2 \sin\theta \, d\theta
\end{equation*}
L'angolo solido è quindi:
\begin{equation*}
    d\Omega = \frac{dS}{r^2} = 2\pi \sin\theta \, d\theta \quad \Rightarrow \quad d\sigma \propto d\Omega
\end{equation*}

\noindent
\textbf{DEF. SEZIONE D'URTO DIFFERENZIALE:} $\Rightarrow$ Sez. d'urto per unità di angolo solido
\begin{equation}
    \boxed{ D(\theta) = \frac{d\sigma}{d\Omega} }
\end{equation}

\noindent
Siano $N_0$ le particelle incidenti e $R$ il numero di particelle rimosse (scatterate). Se $n$ è la densità dei centri di scattering, abbiamo:
\begin{equation*}
    R = n \sigma N_0 \quad \Rightarrow \quad dN = n \, d\sigma \, N_0 \quad \Rightarrow \quad \frac{dN}{d\Omega} = n \frac{d\sigma}{d\Omega} N_0
\end{equation*}
Da cui si ricava operativamente la sezione d'urto differenziale:
\begin{equation}
    \boxed{ \frac{d\sigma}{d\Omega} = D(\theta) = \frac{1}{n N_0} \frac{dN}{d\Omega} }
\end{equation}
(Questa formula correla il numero di particelle scatterate a un determinato angolo solido).

Sebbene Rutherford spieghi lo scattering (e in linea di principio l'emissione di luce):
\begin{itemize}
    \item \textbf{Teorema di Earnshaw:} una collezione di cariche in cui agisce solo $\vec{F}_{el}$ non può stare in equilibrio stabile. $\Rightarrow$ QUINDI L'ELETTRONE DEVE GIRARE.
    \item Se gira, deve subire un'accelerazione normale (centripeta).
    \item Per l'elettromagnetismo classico, una carica accelerata \textbf{EMETTE LUCE SEMPRE} (irradia energia).
    \item Quindi l'elettrone perde energia, inizia un moto a spirale e \textbf{collassa sul nucleo} in $\tau \simeq 10^{-12} \text{ s}$! (Problema di stabilità).
    \item Inoltre, spiraleggiando, la luce emessa avrebbe una frequenza in continuo cambiamento, portando a uno \textbf{spettro a banda continua} (in totale contrasto con i dati sperimentali che mostrano righe spettrali discrete).
\end{itemize}

\subsection{Sorgenti Termiche}

\begin{figure}[H]

    \begin{minipage}{0.65\textwidth}
        Un corpo con temperatura $T > T_{amb}$ vuole andare in equilibrio termico a $T_{amb} \Rightarrow$ l'emissione di luce lo raffredda.
        
        \textit{Esempi pratici:}
        \begin{itemize}
            \item Rame (Cu) a $T = \text{600°C} \Rightarrow$ Colore rosso/arancione (luce visibile).
            \item Corpo umano a $T = \text{310 K} \Rightarrow$ Emette a $\lambda \simeq 10 \mu\text{m}$ (Infrarosso - IR).
        \end{itemize}

\noindent
Definiamo la potenza emessa da un corpo per unità di superficie tra le frequenze $\nu$ e $\nu + d\nu$:
\begin{equation}
    dW = e(\nu, T) \, d\nu \quad [\text{W/m}^2]
\end{equation}
dove $e(\nu, T)$ è il \textbf{Potere emissivo} (o emissività) a temperatura fissata, misurato in $[\text{J/m}^2]$.

\noindent
La potenza totale irradiata su tutta la superficie e su tutte le frequenze è:
\begin{equation}
    W_{TOT} = \int_{S_{corpo}} dS \int_0^{+\infty} e(\nu, T) \, d\nu
\end{equation}\\

\noindent
Definiamo inoltre il \textbf{Potere assorbente} $a(\nu, T)$ come la frazione di potenza assorbita dal corpo. È una funzione adimensionale tale che $a \le 1$.
Per la conservazione dell'energia, la somma delle frazioni assorbita ($a$), riflessa ($r$) e trasmessa ($t$) deve essere unitaria:
\begin{equation}
    a(\nu_0, T_0) + r(\nu_0, T_0) + t(\nu_0, T_0) = 1
\end{equation}

\subsection{Teorema di Kirchhoff (1870)}

Kirchhoff dimostrò che il rapporto tra potere emissivo e potere assorbente è una \textbf{funzione universale} della frequenza e della temperatura, indipendente dal materiale:
\begin{equation}
    \boxed{ \frac{e(\nu, T)}{a(\nu, T)} = f(\nu, T) }
\end{equation}
Fissando $\nu_0$ e $T_0$, si ottiene $f_0 = \frac{e(\nu_0, T_0)}{a(\nu_0, T_0)}$. 
Questo significa che un corpo \textbf{EMETTE BENE dove ASSORBE} bene.

\section{Definizione di Corpo Nero} %

\textbf{DEFINIZIONE:} Un corpo nero è un oggetto ideale che assorbe tutta la radiazione elettromagnetica incidente, indipendentemente dalla frequenza e dalla temperatura. %
\begin{equation*}
    a_{CN}(\nu, T) = 1 \quad \forall \nu, T \quad \Rightarrow \quad \boxed{ e_{CN}(\nu, T) = f(\nu, T) }
\end{equation*}

\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}

    \begin{minipage}{0.65\textwidth}

        Al contempo, le pareti interne della cavità emettono luce. %

\subsection{Densità di Energia Spettrale} %

\begin{figure}[H]

        \begin{equation*}
            e_{\text{CORPO}}(\nu, T) = a_{\text{CORPO}}(\nu, T) \cdot e_{CN}(\nu, T)
        \end{equation*}
        Poiché $a_{\text{CN}} = 1$ quindi si ha che $e_{\text{CN}} = f(\nu, T)$. %
    \end{minipage}
\end{figure}

\begin{figure}[H]
    \centering
    \begin{minipage}{0.25\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-03-05-18-22-34.png}
    \end{minipage}\hfill
    \begin{minipage}{0.73\textwidth}
        Definiamo $\mu(\nu, T)$ come la \textbf{Densità di Energia Spettrale} $[\text{J/m}^3]$. %
        $\mu(\nu, T) d\nu$ rappresenta l'energia per unità di volume compresa nel range di frequenze tra $\nu$ e $\nu + d\nu$. %
        
        Consideriamo un volumetto cilindrico di luce di area di base $S$ che avanza a velocità $c$ in un tempo $dt$ (ovvero di lunghezza $dz = c dt$). L'energia elettromagnetica contenuta è: %
        \begin{equation*}
            dU = \mu \, dz \, S = \mu \, c \, dt \, S
        \end{equation*}
        La potenza irradiata per unità di superficie nel range di frequenze considerato è: %
        \begin{equation*}
            dW = \frac{dU_{d\nu}}{dt \, S} = \frac{\mu(\nu, T) d\nu \cdot S \cdot c \, dt}{S \, dt} = c \, \mu(\nu, T) \, d\nu
        \end{equation*}
        Poiché per definizione $dW = e_{CN}(\nu, T) d\nu$, otteniamo una prima stima utile: %
        \begin{equation*}
            \boxed{ e_{CN} = c \, \mu(\nu, T) }
        \end{equation*}
    \end{minipage}
\end{figure}

Il calcolo precedente prende in cosiderazione solo le onde con direzione di propapagazione perpendicolare a {S}. Tuttavia, facendo i calcoli esatti (mediando su tutte le direzioni spaziali con il Vettore di Poynting), si ottiene un fattore correttivo geometrico pari a $1/4$: %
\begin{equation}
    e_{CN}(\nu, T) = \frac{c}{4} \mu(\nu, T)
\end{equation}

\noindent
L'obiettivo pratico diventa quindi \textbf{CALCOLARE $\mu(\nu, T)$} per poi \textbf{STIMARE $e(\nu, T)$} teorica, e fare il \textbf{CONFRONTO CON I DATI} sperimentali. %
La formula base per ricavare la densità di energia spettrale è il prodotto tra numero di onde e la loro energia: %
\begin{equation}
    \mu(\nu, T) d\nu = \frac{1}{V} \times dN_{\text{ONDE}} \times \langle E \rangle_{\text{1 ONDA}} \quad \text{(valido tra } \nu \text{ e } \nu + d\nu \text{)}
\end{equation}

        Consideriamo un'onda elettromagnetica piana (es. la componente $y$ del campo elettrico) che si propaga lungo l'asse $x$: %
        \begin{equation*}
            \vec{\mathcal{E}}_y(x,t) = \mathcal{E}_{01} \hat{u}_y e^{-i(k_x x - \omega t)} + \mathcal{E}_{02} \hat{u}_y e^{i(k_x x + \omega t)}

Applicando la prima condizione al contorno sulla parete metallica in $x=0$, il campo elettrico tangenziale deve necessariamente annullarsi: %
\begin{equation*}
    \vec{\mathcal{E}}_y(0,t) = 0 \quad \Rightarrow \quad \mathcal{E}_{01} \hat{u}_y e^{i\omega t} + \mathcal{E}_{02} \hat{u}_y e^{i\omega t} = 0 \quad \Rightarrow \quad \mathcal{E}_{01} = -\mathcal{E}_{02}
\end{equation*}

\noindent
Sostituendo $\mathcal{E}_{02}$ e raccogliendo i termini, otteniamo l'equazione caratteristica di un'\textbf{onda stazionaria}: %
\begin{equation*}
    \vec{\mathcal{E}}_y(x,t) = \mathcal{E}_{01} \hat{u}_y e^{i\omega t} \left[ e^{ik_x x} - e^{-ik_x x} \right] = \mathcal{E}_{01} \hat{u}_y e^{i\omega t} [2i \sin(k_x x)]
\end{equation*}
Applicando la seconda condizione al contorno sulla parete opposta in $x=L$: %
\begin{equation*}
    \vec{\mathcal{E}}_y(L,t) = 0 \quad \Rightarrow \quad \sin(k_x L) = 0
\end{equation*}
Affinché il seno sia nullo, l'argomento deve essere per forza un multiplo intero di $\pi$: %
\begin{equation*}
    k_x L = n_x \pi \quad \Rightarrow \quad k_x = n_x \frac{\pi}{L} \quad (\text{con } n_x = 1, 2, 3 \dots)
\end{equation*}
Generalizzando lo stesso ragionamento in tre dimensioni, le componenti del vettore d'onda $\vec{k}$ risultano quantizzate: %
\begin{equation}
    \begin{cases}
        k_x = n_x \frac{\pi}{L} \\
        k_y = n_y \frac{\pi}{L} \\
        k_z = n_z \frac{\pi}{L}
    \end{cases} \quad \Rightarrow \quad \text{definiti POSITIVI}
\end{equation}

        Per calcolare l'energia media di un'onda, la associamo all'energia di un \textbf{oscillatore armonico} (immaginiamo le pareti della cavità composte da atomi che si comportano come oscillatori in equilibrio termico con la radiazione). %
        \begin{equation*}
            \langle E \rangle_{\text{1 ONDA}} = \langle E \rangle_{\text{OSC}} \quad \text{dove} \quad E_{\text{OSC}} = \frac{1}{2}mv^2 + \frac{1}{2}Kx^2
        \end{equation*} %

Per il Teorema di Equipartizione dell'energia: %
\begin{equation*}
    \langle E \rangle_{\text{OSC}} = \frac{1}{2}k_B T + \frac{1}{2}k_B T = k_B T
\end{equation*} %
Calcoliamo ora la densità di energia spettrale $\mu(\nu, T)$. % 
\textit{Nota cruciale:} per ogni direzione di propagazione ci sono $\textcolor{red}{2}$ direzioni indipendenti di polarizzazione dell'onda elettromagnetica. Dobbiamo quindi moltiplicare gli stati per 2: %

\begin{align*}
    \mu(\nu, T) \, d\nu &= \frac{1}{V} \times \left( \textcolor{red}{2} \times dN_{\text{ONDE}} \right) \times \langle E \rangle_{\text{1 ONDA}} = \\
    &= \frac{1}{L^3} \times \left[ \textcolor{red}{2} \times \left( \frac{4\pi L^3}{c^3} \nu^2 \, d\nu \right) \right] \times (k_B T) = \\
    &= \frac{8\pi \nu^2}{c^3} k_B T \, d\nu
\end{align*} %

Questa è la \textbf{Formula di Rayleigh-Jeans} per il corpo nero: %
\begin{equation}
    \boxed{ \mu(\nu, T) = \frac{8\pi \nu^2}{c^3} k_B T }
\end{equation} %
$\Rightarrow$ Questa formula \textbf{NON È IN ACCORDO CON I DATI SPERIMENTALI}. \\

\noindent
Se proviamo a calcolare la potenza totale irraggiata integrando su tutte le frequenze, l'integrale \textbf{diverge} perché la densità cresce con il quadrato della frequenza ($\mu \propto \nu^2$): %
\begin{equation}
    W_{\text{TOT}} = S \int_0^{+\infty} e_{CN}(\nu, T) d\nu = S \frac{c}{4} \int_0^{+\infty} \mu(\nu, T) d\nu \rightarrow +\infty
\end{equation} %

Questo assurdo fisico prende il nome di \textbf{CATASTROFE ULTRAVIOLETTA}. \\

\section{L'Ipotesi di Planck (1900)} %

L'ipotesi fondamentale di Planck è che l'energia scambiata con la materia sia quantizzata: %
\begin{equation}
    \boxed{ E = n(h\nu) } \quad \text{con } n = 0, 1, 2, 3, \dots %
\end{equation}

\noindent
Calcoliamo l'energia media dell'oscillatore $\langle E \rangle_{osc}$ usando la distribuzione statistica di Boltzmann (Planck utilizza ancora le formule statistiche classiche): %
\begin{equation*}
    \langle G \rangle = \frac{\int G(q,p) \rho(q,p) \, dq \, dp}{\int \rho(q,p) \, dq \, dp} \quad \Rightarrow \quad \langle E \rangle_{osc} = \frac{\sum_{n=0}^{+\infty} n h \nu e^{-\frac{n h \nu}{k_B T}}}{\sum_{n=0}^{+\infty} e^{-\frac{n h \nu}{k_B T}}} %
\end{equation*}


Risolviamo separatamente numeratore e denominatore. 

\textbf{Denominatore (1):} %
\begin{equation*}
    \sum_{n=0}^{+\infty} e^{-\frac{n h \nu}{k_B T}} = \left\{ x = e^{-\frac{h \nu}{k_B T}} \text{ con } |x| < 1 \right\} = \sum_{n=0}^{+\infty} x^n = \frac{1}{1-x} %
\end{equation*}

\textbf{Numeratore (2):} %
\begin{align*}
    \sum_{n=0}^{+\infty} n h \nu e^{-\frac{n h \nu}{k_B T}} &= h\nu e^{-\frac{h\nu}{k_B T}} + 2h\nu e^{-\frac{2h\nu}{k_B T}} + 3h\nu e^{-\frac{3h\nu}{k_B T}} + \dots = \\
    &= h\nu e^{-\frac{h\nu}{k_B T}} \left[ 1 + 2e^{-\frac{h\nu}{k_B T}} + 3e^{-\frac{2h\nu}{k_B T}} + \dots \right] = \\
    &= \left\{ x = e^{-\frac{h\nu}{k_B T}} \right\} = h\nu x [1 + 2x + 3x^2 + \dots] = \\
    &= h\nu x \left[ \sum_{n=0}^{+\infty} \frac{d}{dx}(x^n) \right] = h\nu x \left[ \frac{1}{(1-x)^2} \right] %
\end{align*}

Mettendo insieme i due risultati: %
\begin{align*}
    \langle E \rangle_{osc} &= \frac{\frac{h\nu x}{(1-x)^2}}{\frac{1}{1-x}} = \frac{h\nu x}{1-x} = \frac{h\nu e^{-\frac{h\nu}{k_B T}}}{1 - e^{-\frac{h\nu}{k_B T}}} \cdot \frac{e^{\frac{h\nu}{k_B T}}}{e^{\frac{h\nu}{k_B T}}} \quad \Rightarrow \quad \boxed{ \langle E \rangle_{osc} = \frac{h\nu}{e^{\frac{h\nu}{k_B T}} - 1} } %
\end{align*}
(Nota: Possiamo anche esprimerla come $\langle E \rangle_{osc} = \langle n \rangle h\nu$, da cui si ricava il numero medio dei quanti di energia: $\langle n \rangle = \frac{1}{e^{\frac{h\nu}{k_B T}} - 1}$). \\

\noindent
Sostituendo l'energia media corretta $\langle E \rangle_{osc}$ nella formula della densità degli stati calcolata in precedenza, otteniamo la \textbf{Formula di Planck per il corpo nero}: %
\begin{equation}
    \boxed{ \mu(\nu, T) = \frac{8\pi}{c^3} \frac{h\nu^3}{e^{\frac{h\nu}{k_B T}} - 1} } %
\end{equation}
(Questa formula riproduce perfettamente i risultati di laboratorio). \\

\noindent
\textbf{Verifica dei limiti:} %
\begin{enumerate}
    \item \textbf{Basse frequenze ($\nu \to 0$):} Sviluppando l'esponenziale in serie di Taylor ($e^x \approx 1 + x$): %
    \begin{equation*}
        \mu(\nu, T) \simeq \frac{8\pi}{c^3} \frac{h\nu^3}{1 + \frac{h\nu}{k_B T} - 1} = \frac{8\pi \nu^2}{c^3} (k_B T) \quad \rightarrow \quad \text{\textbf{Formula Classica}} %
    \end{equation*}
    \item \textbf{Alte frequenze ($\nu \to \infty$):} L'esponenziale domina ($e^{\frac{h\nu}{k_B T}} \gg 1$): %
    \begin{equation*}
        \mu(\nu, T) \to 0 \quad \text{(La catastrofe ultravioletta è risolta!)} %
    \end{equation*}
\end{enumerate}

\subsection{Derivazione delle Leggi Empiriche} %

\textbf{1. Potenza totale (Stefan-Boltzmann):} %
Calcoliamo la potenza totale emessa $W_{TOT}$ integrando su tutte le frequenze: %
\begin{align*}
    W_{TOT} &= \frac{S c}{4} \int_0^{+\infty} \mu(\nu, T) \, d\nu = \frac{S c}{4} \int_0^{+\infty} \frac{8\pi}{c^3} \frac{h\nu^3}{e^{\frac{h\nu}{k_B T}} - 1} \, d\nu = \\
    &= S \cdot \underbrace{\left[ \frac{2\pi^5}{15} \frac{k_B^4}{h^3 c^2} \right]}_{=\sigma} T^4 = S \sigma T^4 %
\end{align*}

\noindent
\textbf{2. Legge dello Spostamento di Wien:} %
Passando alla distribuzione in lunghezza d'onda $\mu(\lambda, T)$ e imponendo la derivata nulla per trovare il picco: %
\begin{equation*}
    \frac{d\mu(\lambda, T)}{d\lambda} = 0 \quad \Rightarrow \quad \text{Trovo } \lambda_{max} \quad \Rightarrow \quad \lambda_{max} T = \frac{hc}{5 k_B} %
\end{equation*}

\noindent
\textbf{Determinazione delle costanti:} %
Poiché $\sigma$ e il prodotto $\lambda_{max}T$ sono misurabili sperimentalmente (sono noti dai dati), abbiamo un sistema di due equazioni con due incognite ($h$ e $k_B$): %
\begin{equation}
    \begin{cases}
        \frac{2\pi^5 k_B^4}{15 h^3 c^2} = \sigma \\
        \frac{hc}{5 k_B} = \lambda_{max} T
    \end{cases} \quad \Rightarrow \quad \boxed{ \text{Si ricavano i valori di } h \text{ e } k_B } %
\end{equation}

Questo rapporto è un numero \textbf{gigantesco in senso negativo}. Significa che, in condizioni ambientali, praticamente TUTTI gli elettroni dell'intera collezione si trovano nello \textbf{STATO FONDAMENTALE ($n=1$)} %.

QUINDI la Serie di Balmer (che comporta assorbimenti o emissioni da/verso $n=2$) si vede \textbf{solo in Emissione} (dopo aver preventivamente eccitato il gas ad altissime temperature o con scariche elettriche). Non ci sono atomi che possono assorbire fotoni visibili partendo da $n=2$! %
Per eccitarli dallo stato fondamentale, serve la luce UV (che attiva la Serie di Lyman) %.

\section{Regole di Quantizzazione di Wilson e Sommerfeld} %

Osservando i risultati precedenti: %
\begin{equation*}
    \left.
    \begin{aligned}
        &\text{\textbf{Planck}: Osc. ARMONICO} \Rightarrow E = n h \nu \\ %
        &\text{\textbf{Einstein}: } E = h \nu \\ %
        &\text{\textbf{Bohr}: } L = n \hbar %
    \end{aligned}
    \right\} \quad \text{c'è una regola generale} %
\end{equation*}

\noindent
Wilson e Sommerfeld si sono concentrati sui \textbf{SISTEMI PERIODICI} (Einstein escluso). %

\noindent
Per ogni coordinata generalizzata $(q_i, p_i)$ dello spazio delle fasi, postularono la quantizzazione dell'\textbf{Azione}: %
\begin{equation}
    \boxed{ \oint p_i \, dq_i = n_i h } \quad \Rightarrow \quad \text{QUANTIZZAZIONE DELL'AZIONE} %
\end{equation}

\subsection{Applicazione: Oscillatore Armonico} %

L'Hamiltoniana di un oscillatore armonico unidimensionale è: %
\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-03-13-15-24-47.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        \begin{equation*}
            H = \frac{p^2}{2m} + \frac{1}{2} m \omega^2 q^2 = E = \text{cost} %
        \end{equation*}
        
        L'equazione dell'energia rappresenta un'ellisse nello spazio delle fasi. Riscriviamola nella forma canonica $\frac{p^2}{a^2} + \frac{q^2}{b^2} = 1$: %
        \begin{equation*}
            \frac{p^2}{(\sqrt{2mE})^2} + \frac{q^2}{\left(\sqrt{\frac{2E}{m\omega^2}}\right)^2} = 1 %
        \end{equation*}
    \end{minipage}
\end{figure}
Identifichiamo i semiassi: $a = \sqrt{2mE}$ e $b = \sqrt{\frac{2E}{m\omega^2}}$. %

L'integrale di linea chiuso $\oint p \, dq$ corrisponde geometricamente all'area $A$ racchiusa dall'ellisse: %
\begin{equation*}
    A = \pi a b = \pi (\sqrt{2mE}) \left(\sqrt{\frac{2E}{m\omega^2}}\right) = \pi \sqrt{\frac{4 m E^2}{m \omega^2}} = \frac{2\pi E}{\omega} %
\end{equation*}

Applicando la regola di quantizzazione di Wilson-Sommerfeld ($\oint p \, dq = nh$): %
\begin{equation*}
    \frac{2\pi E}{\omega} = n h \quad \Rightarrow \quad E = n \frac{h}{2\pi} \omega %
\end{equation*}
Riconoscendo che $\hbar = \frac{h}{2\pi}$ e $\omega = 2\pi\nu$, otteniamo: %
\begin{equation}
    \boxed{ E = n \hbar \omega = n h \nu } %
\end{equation}
\textit{Siamo tornati all'ipotesi iniziale di Planck!} %

\subsection{Il Limite Classico ($n \to +\infty$)} %

Valutiamo la differenza di energia relativa tra due livelli adiacenti per grandi numeri quantici: %
\begin{equation*}
    \frac{E_{n+1} - E_n}{E_n} = \frac{(n+1)\hbar\omega - n\hbar\omega}{n\hbar\omega} = \frac{\hbar\omega}{n\hbar\omega} = \frac{1}{n} %
\end{equation*}
Prendendo il limite per $n \to +\infty$: %
\begin{equation*}
    \lim_{n \to \infty} \frac{1}{n} = 0 \quad \Rightarrow \quad 0 %
\end{equation*}
Questo significa che se $n$ è molto grande (macroscopico), la distanza fra i salti di quantizzazione diventa trascurabile: l'energia $E$ torna a comportarsi come un continuo %. 
Si ritrova così il $\Rightarrow$ \textbf{LIMITE CLASSICO} %.

\subsection{Esempio Macroscopico del Limite Classico} %

Consideriamo un oscillatore armonico macroscopico (un tipico sistema massa-molla): %

Ipotizziamo i seguenti valori realistici di laboratorio: %
\begin{itemize}
    \item $K = 1 \text{ N/m}$ e $m = 1 \text{ kg}$ %
    \item $x_0 = 10 \text{ cm} = 0.1 \text{ m}$ (posizione iniziale) %
    \item $v_0 = 1 \text{ m/s}$ (velocità iniziale) %
\end{itemize}

L'energia totale $E_{TOT}$ del sistema classico è: %
\begin{equation*}
    E_{TOT} = \frac{1}{2} m v_0^2 + \frac{1}{2} K x_0^2 \simeq 0.5 \text{ J} %
\end{equation*}
La pulsazione angolare $\omega$ è data da: %
\begin{equation*}
    \omega = \sqrt{\frac{K}{m}} = 1 \text{ rad/s} %
\end{equation*}

Calcoliamo l'energia del singolo "quanto" per questo specifico oscillatore: %
\begin{equation*}
    E = \hbar \omega = 1.05 \cdot 10^{-34} \text{ J} %
\end{equation*}

Il numero di quanti $n$ che compongono l'energia totale è quindi: %
\begin{equation*}
    n = \frac{E_{TOT}}{E} \cong 10^{33} \quad \text{(numero di quanti)} %
\end{equation*}
Essendo il numero di quanti così spaventosamente grande, $\Rightarrow$ \textbf{NON ci accorgiamo della quantizzazione}. L'energia ci appare a tutti gli effetti come un \textbf{continuo} %.

\section{Problemi Aperti della Teoria di Bohr-Sommerfeld} %

Sebbene le regole di quantizzazione dell'azione forniscano risultati coerenti per alcuni sistemi specifici, la teoria presenta dei limiti insormontabili: %

\begin{itemize}
    \item $\rightarrow$ \textbf{Vale solo per sistemi periodici} %.
    \item $\rightarrow$ \textbf{Non prevede quali righe} si vedono negli spettri atomici (quali transizioni sono permesse o "proibite") %.
    \begin{itemize}
        \item $\hookrightarrow$ E \textbf{non ne prevede l'intensità} (quanto sono brillanti le righe emesse) %.
        \item $\hookrightarrow$ \textbf{Non prevede la presenza di più elettroni} (il modello fallisce per atomi con $Z > 1$, incapace di descrivere le interazioni repulsive tra gli elettroni stessi) %.
    \end{itemize}
\end{itemize}

\subsection{Derivazione del Postulato di Bohr da Wilson-Sommerfeld} %

\begin{figure}[H]
    \centering
    \begin{minipage}{0.3\textwidth}
        \centering
        \includegraphics[width=\linewidth]{immagini/2026-03-21-14-27-16.png}
    \end{minipage}\hfill
    \begin{minipage}{0.65\textwidth}
        Possiamo dimostrare come il postulato di Bohr sia in realtà un caso particolare della regola di Wilson-Sommerfeld applicata a un'orbita circolare %.
        Scegliamo come coordinate generalizzate l'angolo $\theta$ e il momento angolare $L$: %
    \end{minipage}
\end{figure}
\begin{equation*}
    q \rightarrow \theta \quad , \quad p \rightarrow L %
\end{equation*}

Applicando la regola di quantizzazione dell'azione sull'intero periodo (un giro completo): %
\begin{equation*}
    \oint p \, dq = \oint L \, d\theta %
\end{equation*}
Poiché in un'orbita circolare in un campo di forze centrali il momento angolare $L$ si conserva, esso è una costante e può essere portato fuori dall'integrale: %
\begin{equation*}
    L \oint d\theta = L (2\pi) %
\end{equation*}
Uguagliando l'espressione a $nh$: %
\begin{equation}
    2\pi L = n h \quad \Rightarrow \quad \boxed{ L = n \frac{h}{2\pi} = n\hbar } \quad \Downarrow \text{ Sistemi periodici} %
\end{equation}
Abbiamo così riottenuto in modo formale la condizione di quantizzazione del momento angolare imposta da Bohr! %
